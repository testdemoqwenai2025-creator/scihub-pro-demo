CTA #8: "Join Waitlist" (Email Form Submit)
┌── Current Behavior ──────────────────────────────────────────────────────────┐
│ • Location: Line ~734-735                                                    │
│ • Current: Inside <form> tag but no onSubmit handler                          │
│ • Has email input field                                                      │
└──────────────────────────────────────────────────────────────────────────────┘
┌── REQUIRED BEHAVIOR ─────────────────────────────────────────────────────────┐
│ ACTION: Capture Email & Add to Waitlist                                      │
│                                                                              │
│ IMPLEMENTATION:                                                              │
│ □ Add onSubmit handler to form:                                              │
│                                                                              │
│   const handleWaitlistSubmit = async (e) => {                                │
│     e.preventDefault();                                                      │
│     const email = e.target.email.value;                                      │
│                                                                              │
│     // Validate email format                                                 │
│     if (!email.includes('@')) {                                             │
│       setError('Please enter a valid email');                                │
│       return;                                                                │
│     }                                                                       │
│                                                                              │
│     setLoading(true);                                                        │
│                                                                              │
│     try {                                                                   │
│       // Call waitlist API                                                   │
│       const res = await fetch('/api/waitlist', {                             │
│         method: 'POST',                                                     │
│         body: JSON.stringify({ email, source: 'landing_page' })             │
│       });                                                                  │
│                                                                              │
│       if (res.ok) {                                                         │
│         setSuccess(true); // Show success state                              │
│         // Confetti animation optional                                       │
│       } else {                                                              │
│         setError('Something went wrong. Please try again.');                 │
│       }                                                                     │
│     } catch {                                                               │
│       // Fallback: store locally, sync later                                 │
│       localStorage.setItem('scihub_waitlist_email', email);                  │
│       setSuccess(true); // Optimistic update                                 │
│     } finally {                                                             │
│       setLoading(false);                                                     │
│     }                                                                       │
│   };                                                                        │
│                                                                              │
│ □ Create /api/waitlist route if not exists:                                  │
│   - Store emails in array/file                                              │
│   - Deduplicate by email                                                    │
│   - Return success/error                                                    │
│   - Rate limit (max 5 submissions/hour per IP)                              │
│                                                                              │
│ □ UI States:                                                                │
│   - Default: Email input + "Join Waitlist" button                            │
│   - Loading: Spinner on button, disabled input                               │
│   - Success: Green checkmark + "You're on the list! 🎉"                     │
│   - Error: Red message + retry option                                       │
│                                                                              │
│ □ Bonus: Add checkbox "I want early access" vs "Just updates"               │
└──────────────────────────────────────────────────────────────────────────────┘

COMPONENT IMPLEMENTATION:
```tsx
// components/ComingSoon.tsx
interface ComingSoonProps {
  featureName: string;
  benefits: string[];
  estimatedDate: string;
  progress: number; // 0-100
  alternativeLink?: { label: string; href: string };
  notifyApiEndpoint: string;
}

export function ComingSoon({ 
  featureName, 
  benefits, 
  estimatedDate, 
  progress,
  alternativeLink,
  notifyApiEndpoint 
}: ComingSoonProps) {
  const [email, setEmail] = useState('');
  const [notified, setNotified] = useState(false);
  const [loading, setLoading] = useState(false);

  const handleNotify = async () => {
    setLoading(true);
    try {
      await fetch(notifyApiEndpoint, {
        method: 'POST',
        body: JSON.stringify({ email, feature: featureName })
      });
      setNotified(true);
    } catch {
      // Optimistic update
      setNotified(true);
    }
    setLoading(false);
  };

  return (
    <Dialog>
      <DialogContent>
        <DialogTitle>🚧 {featureName}</DialogTitle>
        <div className="space-y-4">
          <p className="text-muted-foreground">
            This feature is under development.
          </p>
          
          <ul className="space-y-2">
            {benefits.map((b, i) => (
              <li key={i} className="flex items-center gap-2">
                <span className="text-green-500">✓</span> {b}
              </li>
            ))}
          </ul>

          <div>
            <p className="text-sm font-medium mb-2">Progress</p>
            <Progress value={progress} />
            <p className="text-xs text-muted-foreground mt-1">
              Estimated: {estimatedDate}
            </p>
          </div>

          {!notified ? (
            <div className="space-y-2">
              <Input
                placeholder="your@email.com"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
              />
              <Button 
                onClick={handleNotify} 
                disabled={!email || loading}
                className="w-full"
              >
                {loading ? '...' : 'Notify Me'}
              </Button>
            </div>
          ) : (
            <div className="bg-green-50 p-3 rounded-lg text-green-700 text-sm">
              ✓ We'll notify you when it's ready!
            </div>
          )}

          {alternativeLink && (
            <p className="text-sm text-center">
              Try <Link href={alternativeLink.href}>{alternativeLink.label}</Link>
            </p>
          )}
        </div>
      </DialogContent>
    </Dialog>
  );
}
```
