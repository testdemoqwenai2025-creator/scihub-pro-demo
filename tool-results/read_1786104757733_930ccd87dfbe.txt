     2→
     3→/**
     4→ * SciHub Pro - Collaboration Page
     5→ * 
     6→ * Team collaboration features with:
     7→ * - Project management

     9→ * - Discussion forums
    10→ * - Real-time editing stubs
    11→ * - Activity feeds

    14→
    15→import { useState, useEffect } from 'react';
    16→import { useTranslation } from '@/i18n/useTranslation';
    17→import { useSciHubStore, createDynamicField } from '@/store/useSciHubStore';
    18→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    19→import { Badge } from '@/components/ui/badge';
    20→import { Button } from '@/components/ui/button';
    21→import { Input } from '@/components/ui/input';
    22→import { Textarea } from '@/components/ui/textarea';
    23→import {
    24→  Select,
    25→  SelectContent,
    26→  SelectItem,
    27→  SelectTrigger,
    28→  SelectValue,
    29→} from '@/components/ui/select';
    30→import {
    31→  Dialog,
    32→  DialogContent,
    33→  DialogHeader,
    34→  DialogTitle,
    35→  DialogTrigger,
    36→} from '@/components/ui/dialog';
    37→import {
    38→  Tabs,
    39→  TabsContent,
    40→  TabsList,
    41→  TabsTrigger,
    42→} from '@/components/ui/tabs';
    43→import { CollaborationSkeleton, ListSkeleton, CardSkeleton } from '@/components/SkeletonComponents';
    44→
    45→// ============ COLLABORATION PAGE COMPONENT ============
    46→
    47→export default function CollaborationPage() {
    48→  const { t } = useTranslation();
    49→  const store = useSciHubStore();
    50→  
    51→  const {
    52→    projects,
    53→    teamMembers,
    54→    discussions,
    55→    addProject,
    56→    updateProject,
    57→    addDiscussion,
    58→    addReply,
    59→    activities,
    60→    addActivity,
    61→    notifications,
    62→    addNotification,
    63→    userProfile,
    64→    upgradePrompts,
    65→    triggerUpgradePrompt,
    66→  } = store;
    67→
    68→  // UI State
    69→  const [activeTab, setActiveTab] = useState('projects');
    70→  const [showNewProjectDialog, setShowNewProjectDialog] = useState(false);
    71→  const [showNewDiscussionDialog, setShowNewDiscussionDialog] = useState(false);
    72→  const [selectedProject, setSelectedProject] = useState<string | null>(null);
    73→  const [replyTexts, setReplyTexts] = useState<Record<string, string>>({});
    74→  
    75→  // New project form state
    76→  const [newProjectName, setNewProjectName] = useState('');
    77→  const [newProjectDesc, setNewProjectDesc] = useState('');
    78→  const [newProjectVisibility, setNewProjectVisibility] = useState<'public' | 'private' | 'team'>('team');
    79→  
    80→  // New discussion form state
    81→  const [newDiscussionTitle, setNewDiscussionTitle] = useState('');
    82→  const [newDiscussionContent, setNewDiscussionContent] = useState('');
    83→  const [newDiscussionTags, setNewDiscussionTags] = useState('');
    84→  const [isLoading, setIsLoading] = useState(true);
    85→
    86→  // Simulate loading on mount
    87→  useEffect(() => {
    88→    const timer = setTimeout(() => setIsLoading(false), 1500);
    89→    return () => clearTimeout(timer);
    90→  }, []);
    91→
    92→  // Get selected project data
    93→  const selectedProjectData = projects.find(p => p.id === selectedProject);
    94→
    95→  // Get online members count
    96→  const onlineCount = teamMembers.filter(m => m.online).length;
    97→
    98→  // ============ HANDLERS ============
    99→
   100→  const handleCreateProject = () => {
   101→    if (!newProjectName.trim()) return;
   102→
   103→    addProject({
   104→      name: createDynamicField(newProjectName),
   105→      description: createDynamicField(newProjectDesc || `Research project: ${newProjectName}`),
   106→      memberIds: [teamMembers[0]?.id || 'member-001'], // Add current user as lead
   107→      status: 'active',
   108→      visibility: newProjectVisibility,
   109→      leadId: teamMembers[0]?.id || 'member-001',
   110→      datasetIds: [],
   111→      queryIds: [],
   112→    });
   113→
   114→    addActivity({
   115→      type: 'collaboration',
   116→      message: createDynamicField(`Created project: ${newProjectName}`),
   117→      icon: '📁',
   118→    });
   119→
   120→    addNotification({
   121→      type: 'collaboration',
   122→      title: 'New Project Created',
   123→      message: `${userProfile.displayName.value} created "${newProjectName}"`,
   124→      priority: 'medium',
   125→    });
   126→
   127→    // Reset form
   128→    setNewProjectName('');
   129→    setNewProjectDesc('');
   130→    setShowNewProjectDialog(false);
   131→
   132→    // Trigger upgrade prompt for team features
   133→    if (teamMembers.length >= 5) {
   134→      triggerUpgradePrompt('collaboration');
   135→    }
   136→  };
   137→
   138→  const handleCreateDiscussion = () => {
   139→    if (!newDiscussionTitle.trim() || !selectedProject) return;
   140→
   141→    addDiscussion({
   142→      title: createDynamicField(newDiscussionTitle),
   143→      content: createDynamicField(newDiscussionContent),
   144→      authorId: teamMembers[0]?.id || 'member-001',
   145→      projectId: selectedProject,
   146→      tags: newDiscussionTags.split(',').map(t => t.trim()).filter(Boolean),
   147→      replies: [],
   148→    });
   149→
   150→    addActivity({
   151→      type: 'collaboration',
   152→      message: createDynamicField(`Started discussion: ${newDiscussionTitle}`),
   153→      icon: '💬',
   154→    });
   155→
   156→    // Reset form
   157→    setNewDiscussionTitle('');
   158→    setNewDiscussionContent('');
   159→    setNewDiscussionTags('');
   160→    setShowNewDiscussionDialog(false);
   161→  };
   162→
   163→  const handleAddReply = (discussionId: string) => {
   164→    const text = replyTexts[discussionId];
   165→    if (!text?.trim()) return;
   166→
   167→    addReply(discussionId, {
   168→      authorId: teamMembers[0]?.id || 'member-001',
   169→      content: text,
   170→    });
   171→
   172→    setReplyTexts({ ...replyTexts, [discussionId]: '' });
   173→    
   174→    addActivity({
   175→      type: 'collaboration',
   176→      message: createDynamicField('Replied to discussion'),
   177→      icon: '💬',
   178→    });
   179→  };
   180→
   181→  const getMemberById = (id: string) => teamMembers.find(m => m.id === id);
   182→
   183→  const getStatusColor = (status: string) => {
   184→    switch (status) {
   185→      case 'active': return 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200';
   186→      case 'paused': return 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-200';
   187→      case 'completed': return 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-200';
   188→      case 'archived': return 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-200';
   189→      default: return 'bg-gray-100 text-gray-800';
   190→    }
   191→  };
   192→
   193→  const formatTimeAgo = (date: Date) => {
   194→    const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   195→    if (seconds < 60) return 'Just now';
   196→    const minutes = Math.floor(seconds / 60);
   197→    if (minutes < 60) return `${minutes}m ago`;
   198→    const hours = Math.floor(minutes / 60);
   199→    if (hours < 24) return `${hours}h ago`;
   200→    const days = Math.floor(hours / 24);
   201→    return `${days}d ago`;
   202→  };
   203→
   204→  // Loading skeleton
   205→  if (isLoading) {
   206→    return (
   207→      <div className="min-h-screen bg-background">
   208→        <CollaborationSkeleton />
   209→      </div>
   210→    );
   211→  }
   212→
   213→  return (
   214→    <div className="min-h-screen bg-background p-6">
   215→      {/* Header */}
   216→      <div className="mb-8">
   217→        <div className="flex items-start justify-between">
   218→          <div>
   219→            <h1 className="text-3xl font-bold text-foreground">
   220→              👥 {t('collaboration.title') || 'Collaboration'}
   221→            </h1>
   222→            <p className="text-muted-foreground mt-1">
   223→              Work together on research projects, share insights, and discuss findings
   224→            </p>
   225→          </div>
   226→
   227→          {/* Quick Stats */}
   228→          <div className="flex items-center gap-4">
   229→            <div className="flex items-center gap-2">
   230→              <span className="w-2 h-2 rounded-full bg-green-500 animate-pulse"></span>
   231→              <span className="text-sm">{onlineCount} online</span>
   232→            </div>
   233→            <Badge variant="secondary">
   234→              {projects.length} projects
   235→            </Badge>
   236→          </div>
   237→        </div>
   238→
   239→        {/* Team Members Preview */}
   240→        <div className="mt-4 flex items-center gap-3">
   241→          <div className="flex -space-x-2">
   242→            {teamMembers.slice(0, 5).map(member => (
   243→              <div
   244→                key={member.id}
   245→                className={`w-9 h-9 rounded-full border-2 border-background flex items-center justify-center text-sm font-medium ${
   246→                  member.online ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'
   247→                }`}
   248→                title={`${member.name}${member.online ? ' (Online)' : ''}`}
   249→              >
   250→                {member.name.split(' ').map(n => n[0]).join('')}
   251→              </div>
   252→            ))}
   253→            {teamMembers.length > 5 && (
   254→              <div className="w-9 h-9 rounded-full border-2 border-background bg-muted flex items-center justify-center text-xs font-medium">
   255→                +{teamMembers.length - 5}
   256→              </div>
   257→            )}
   258→          </div>
   259→          <span className="text-sm text-muted-foreground">
   260→            {teamMembers.length} team members
   261→          </span>
   262→        </div>
   263→      </div>
   264→
   265→      {/* Main Content */}
   266→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   267→        <TabsList className="mb-6">
   268→          <TabsTrigger value="projects">📁 Projects</TabsTrigger>
   269→          <TabsTrigger value="discussions">💬 Discussions</TabsTrigger>
   270→          <TabsTrigger value="team">👥 Team</TabsTrigger>
   271→          <TabsTrigger value="activity">📋 Activity</TabsTrigger>
   272→        </TabsList>
   273→
   274→        {/* PROJECTS TAB */}
   275→        <TabsContent value="projects" className="space-y-6">
   276→          {/* Projects Header */}
   277→          <div className="flex items-center justify-between">
   278→            <h2 className="text-xl font-semibold">Your Projects</h2>
   279→            
   280→            <Dialog open={showNewProjectDialog} onOpenChange={setShowNewProjectDialog}>
   281→              <DialogTrigger asChild>
   282→                <Button>➕ New Project</Button>
   283→              </DialogTrigger>
   284→              <DialogContent>
   285→                <DialogHeader>
   286→                  <DialogTitle>Create New Project</DialogTitle>
   287→                </DialogHeader>
   288→                <div className="space-y-4 pt-4">
   289→                  <div>
   290→                    <label className="text-sm font-medium">Project Name *</label>
   291→                    <Input
   292→                      placeholder="e.g., Oncology Drug Discovery"
   293→                      value={newProjectName}
   294→                      onChange={(e) => setNewProjectName(e.target.value)}
   295→                      className="mt-1"
   296→                    />
   297→                  </div>
   298→                  
   299→                  <div>
   300→                    <label className="text-sm font-medium">Description</label>
   301→                    <Textarea
   302→                      placeholder="What is this project about?"
   303→                      value={newProjectDesc}
   304→                      onChange={(e) => setNewProjectDesc(e.target.value)}
   305→                      className="mt-1"
   306→                      rows={3}
   307→                    />
   308→                  </div>
   309→
   310→                  <div>
   311→                    <label className="text-sm font-medium">Visibility</label>
   312→                    <Select value={newProjectVisibility} onValueChange={(v) => setNewProjectVisibility(v as any)}>
   313→                      <SelectTrigger className="mt-1">
   314→                        <SelectValue />
   315→                      </SelectTrigger>
   316→                      <SelectContent>
   317→                        <SelectItem value="private">🔒 Private (only you)</SelectItem>
   318→                        <SelectItem value="team">👥 Team (invited members)</SelectItem>
   319→                        <SelectItem value="public">🌐 Public (anyone can view)</SelectItem>
   320→                      </SelectContent>
   321→                    </Select>
   322→                  </div>
   323→
   324→                  <div className="flex gap-2 justify-end pt-2">
   325→                    <Button variant="outline" onClick={() => setShowNewProjectDialog(false)}>
   326→                      Cancel
   327→                    </Button>
   328→                    <Button onClick={handleCreateProject} disabled={!newProjectName.trim()}>
   329→                      Create Project
   330→                    </Button>
   331→                  </div>
   332→                </div>
   333→              </DialogContent>
   334→            </Dialog>
   335→          </div>
   336→
   337→          {/* Projects Grid */}
   338→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   339→            {projects.map(project => (
   340→              <Card 
   341→                key={project.id} 
   342→                className={`hover:shadow-lg transition-all cursor-pointer ${
   343→                  selectedProject === project.id ? 'ring-2 ring-primary' : ''
   344→                }`}
   345→                onClick={() => setSelectedProject(project.id)}
   346→              >
   347→                <CardHeader className="pb-3">
   348→                  <div className="flex items-start justify-between">
   349→                    <CardTitle className="text-lg line-clamp-1">
   350→                      {project.name.value}
   351→                    </CardTitle>
   352→                    <Badge className={getStatusColor(project.status)}>
   353→                      {project.status}
   354→                    </Badge>
   355→                  </div>
   356→                  <CardDescription className="line-clamp-2">
   357→                    {project.description.value}
   358→                  </CardDescription>
   359→                </CardHeader>
   360→                
   361→                <CardContent>
   362→                  {/* Project Meta */}
   363→                  <div className="space-y-3 text-sm">
   364→                    <div className="flex items-center gap-4 text-muted-foreground">
   365→                      <span className="flex items-center gap-1">
   366→                        👥 {project.memberIds.length} members
   367→                      </span>
   368→                      <span className="flex items-center gap-1">
   369→                        📊 {project.datasetIds.length} datasets
   370→                      </span>
   371→                      <span className="flex items-center gap-1">
   372→                        🔎 {project.queryIds.length} queries
   373→                      </span>
   374→                    </div>
   375→
   376→                    {/* Lead */}
   377→                    <div className="flex items-center gap-2">
   378→                      <span className="text-xs text-muted-foreground">Lead:</span>
   379→                      <span className="text-xs font-medium">
   380→                        {getMemberById(project.leadId)?.name || 'Unknown'}
   381→                      </span>
   382→                    </div>
   383→
   384→                    {/* Last Activity */}
   385→                    <div className="text-xs text-muted-foreground">
   386→                      Last active: {formatTimeAgo(project.lastActivity)}
   387→                    </div>
   388→
   389→                    {/* Members Avatars */}
   390→                    <div className="flex items-center gap-2 pt-2 border-t">
   391→                      <div className="flex -space-x-1">
   392→                        {project.memberIds.slice(0, 4).map(id => {
   393→                          const member = getMemberById(id);
   394→                          return (
   395→                            <div
   396→                              key={id}
   397→                              className={`w-6 h-6 rounded-full border border-background flex items-center justify-center text-[10px] ${
   398→                                member?.online ? 'bg-green-100' : 'bg-gray-100'
   399→                              }`}
   400→                              title={member?.name}
   401→                            >
   402→                              {member?.name.charAt(0)}
   403→                            </div>
   404→                          );
   405→                        })}
   406→                        {project.memberIds.length > 4 && (
   407→                          <div className="w-6 h-6 rounded-full border border-background bg-muted flex items-center justify-center text-[10px]">
   408→                            +{project.memberIds.length - 4}
   409→                          </div>
   410→                        )}
   411→                      </div>
   412→                      
   413→                      <Badge variant="outline" className="text-xs ml-auto">
   414→                        {project.visibility}
   415→                      </Badge>
   416→                    </div>
   417→                  </div>
   418→                </CardContent>
   419→              </Card>
   420→            ))}
   421→
   422→            {/* Empty State / Create Prompt */}
   423→            {projects.length === 0 && (
   424→              <Card className="col-span-full">
   425→                <CardContent className="py-12 text-center">
   426→                  <span className="text-4xl block mb-3">📁</span>
   427→                  <h3 className="text-lg font-semibold mb-2">No Projects Yet</h3>
   428→                  <p className="text-muted-foreground max-w-md mx-auto mb-4">
   429→                    Create a project to organize your research, collaborate with team members, and track progress.
   430→                  </p>
   431→                  <Button onClick={() => setShowNewProjectDialog(true)}>
   432→                    ➕ Create Your First Project
   433→                  </Button>
   434→                </CardContent>
   435→              </Card>
   436→            )}
   437→          </div>
   438→
   439→          {/* Selected Project Details */}
   440→          {selectedProjectData && (
   441→            <Card className="mt-6">
   442→              <CardHeader>
   443→                <div className="flex items-start justify-between">
   444→                  <div>
   445→                    <CardTitle>{selectedProjectData.name.value}</CardTitle>
   446→                    <CardDescription className="mt-1">
   447→                      {selectedProjectData.description.value}
   448→                    </CardDescription>
   449→                  </div>
   450→                  <Button
   451→                    variant="outline"
   452→                    size="sm"
   453→                    onClick={() => {
   454→                      setSelectedProject(null);
   455→                    }}
   456→                  >
   457→                    ✕ Close
   458→                  </Button>
   459→                </div>
   460→              </CardHeader>
   461→              
   462→              <CardContent>
   463→                <Tabs defaultValue="overview">
   464→                  <TabsList className="mb-4">
   465→                    <TabsTrigger value="overview">Overview</TabsTrigger>
   466→                    <TabsTrigger value="members">Members</TabsTrigger>
   467→                    <TabsTrigger value="datasets">Datasets</TabsTrigger>
   468→                    <TabsTrigger value="discussions">Discussions</TabsTrigger>
   469→                  </TabsList>
   470→
   471→                  <TabsContent value="overview" className="space-y-4">
   472→                    <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   473→                      <div className="p-3 bg-muted/50 rounded-lg text-center">
   474→                        <div className="text-2xl font-bold">{selectedProjectData.memberIds.length}</div>
   475→                        <div className="text-xs text-muted-foreground">Members</div>
   476→                      </div>
   477→                      <div className="p-3 bg-muted/50 rounded-lg text-center">
   478→                        <div className="text-2xl font-bold">{selectedProjectData.datasetIds.length}</div>
   479→                        <div className="text-xs text-muted-foreground">Datasets</div>
   480→                      </div>
   481→                      <div className="p-3 bg-muted/50 rounded-lg text-center">
   482→                        <div className="text-2xl font-bold">{selectedProjectData.queryIds.length}</div>
   483→                        <div className="text-xs text-muted-foreground">Queries</div>
   484→                      </div>
   485→                      <div className="p-3 bg-muted/50 rounded-lg text-center">
   486→                        <div className="text-2xl font-bold">
   487→                          {discussions.filter(d => d.projectId === selectedProject).length}
   488→                        </div>
   489→                        <div className="text-xs text-muted-foreground">Discussions</div>
   490→                      </div>
   491→                    </div>
   492→
   493→                    <Button 
   494→                      variant="outline" 
   495→                      className="w-full"
   496→                      onClick={() => {
   497→                        setActiveTab('discussions');
   498→                        setShowNewDiscussionDialog(true);
   499→                      }}
   500→                    >
   501→                      💬 Start a Discussion
   502→                    </Button>
   503→                  </TabsContent>
   504→
   505→                  <TabsContent value="members">
   506→                    <div className="space-y-2">
   507→                      {selectedProjectData.memberIds.map(id => {
   508→                        const member = getMemberById(id);
   509→                        if (!member) return null;
   510→                        
   511→                        return (
   512→                          <div key={id} className="flex items-center justify-between p-3 rounded-lg hover:bg-muted/50">
   513→                            <div className="flex items-center gap-3">
   514→                              <div className={`w-10 h-10 rounded-full flex items-center justify-center font-medium ${
   515→                                member.online ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'
   516→                              }`}>
   517→                                {member.name.split(' ').map(n => n[0]).join('')}
   518→                              </div>
   519→                              <div>
   520→                                <p className="font-medium">{member.name}</p>
   521→                                <p className="text-sm text-muted-foreground">{member.role}</p>
   522→                              </div>
   523→                            </div>
   524→                            
   525→                            <div className="flex items-center gap-2">
   526→                              {member.online ? (
   527→                                <Badge variant="secondary" className="bg-green-100 text-green-700">Online</Badge>
   528→                              ) : (
   529→                                <span className="text-xs text-muted-foreground">
   530→                                  Active {formatTimeAgo(member.lastActive)}
   531→                                </span>
   532→                              )}
   533→                              <Badge variant="outline">{member.publicationsCount} pubs</Badge>
   534→                            </div>
   535→                          </div>
   536→                        );
   537→                      })}
   538→                    </div>
   539→                  </TabsContent>
   540→
   541→                  <TabsContent value="datasets">
   542→                    <p className="text-muted-foreground text-center py-8">
   543→                      No datasets linked yet. Add datasets from the Data Lake.
   544→                    </p>
   545→                  </TabsContent>
   546→
   547→                  <TabsContent value="discussions">
   548→                    <div className="space-y-2">
   549→                      {discussions
   550→                        .filter(d => d.projectId === selectedProject)
   551→                        .map(discussion => (
   552→                          <div key={discussion.id} className="p-3 rounded-lg hover:bg-muted/50 cursor-pointer">
   553→                            <div className="flex items-center justify-between">
   554→                              <h4 className="font-medium">{discussion.title.value}</h4>
   555→                              <Badge variant="secondary">{discussion.replyCount} replies</Badge>
   556→                            </div>
   557→                            <p className="text-sm text-muted-foreground mt-1 line-clamp-1">
   558→                              {discussion.content.value}
   559→                            </p>
   560→                          </div>
   561→                        ))
   562→                      }
   563→                      
   564→                      {discussions.filter(d => d.projectId === selectedProject).length === 0 && (
   565→                        <div className="text-center py-8">
   566→                          <p className="text-muted-foreground mb-3">No discussions in this project yet.</p>
   567→                          <Button size="sm" onClick={() => setShowNewDiscussionDialog(true)}>
   568→                            Start First Discussion
   569→                          </Button>
   570→                        </div>
   571→                      )}
   572→                    </div>
   573→                  </TabsContent>
   574→                </Tabs>
   575→              </CardContent>
   576→            </Card>
   577→          )}
   578→        </TabsContent>
   579→
