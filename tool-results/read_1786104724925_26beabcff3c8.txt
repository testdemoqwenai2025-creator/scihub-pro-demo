   196→   196→      dataset: '#06b6d4', // cyan
   197→   197→      concept: '#84cc16', // lime

   200→   200→  };

   203→   203→    const icons: Record<string, string> = {
   204→   204→      paper: '📄',
   205→   205→      gene: '🧬',
   206→   206→      compound: '⚗️',
   207→   207→      author: '👤',
   208→   208→      domain: '🏷️',
   209→   209→      dataset: '📊',
   210→   210→      concept: '💡',
   211→   211→    };

   219→   219→
   220→   220→    const ctx = canvas.getContext('2d');
   221→   221→    if (!ctx) return;
   222→   222→
   223→   223→    // Set canvas size
   224→   224→    const rect = canvas.parentElement?.getBoundingClientRect();
   225→   225→    if (rect) {
   226→   226→      canvas.width = rect.width;
   227→   227→      canvas.height = rect.height;
   228→   228→    }
   229→   229→

   231→   231→    ctx.fillStyle = '#ffffff';
   232→   232→    ctx.fillRect(0, 0, canvas.width, canvas.height);

   234→   234→    if (settings.layout === 'force') {

   239→   239→      // Draw edges first (behind nodes)

   241→   241→        filteredEdges.forEach(edge => {
   242→   242→          const sourceNode = filteredNodes.find(n => n.id === edge.source);
   243→   243→          const targetNode = filteredNodes.find(n => n.id === edge.target);
   244→   244→          
   245→   245→          if (sourceNode && targetNode) {
   246→   246→            ctx.beginPath();
   247→   247→            ctx.moveTo(sourceNode.x, sourceNode.y);
   248→   248→            ctx.lineTo(targetNode.x, targetNode.y);
   249→   249→            ctx.strokeStyle = `${edge.strength < 0.5 ? '#d1d5db' : '#9ca3af'}${Math.floor(edge.strength * 255).toString(16).padStart(2, '0')}`;
   250→   250→            ctx.lineWidth = edge.strength * 3;
   251→   251→            ctx.stroke();
   252→   252→          }
   253→   253→        });
   254→   254→      }
   255→   255→
   256→   256→      // Draw nodes

   259→   259→        const isHovered = node.id === hoveredNode;
   260→   260→        const baseSize = settings.nodeSizeBy === 'connections' ? Math.max(15, node.connections * 3) :
   261→   261→                        settings.nodeSizeBy === 'importance' ? 20 + node.importance * 30 : 25;
   262→   262→        const size = baseSize * (isSelected || isHovered ? 1.3 : 1);
   263→   263→        
   264→   264→        // Node circle
   265→   265→        ctx.beginPath();
   266→   266→        ctx.arc(node.x, node.y, size, 0, Math.PI * 2);

   268→   268→        ctx.fill();
   269→   269→        
   270→   270→        if (isSelected) {

   272→   272→          ctx.lineWidth = 3;
   273→   273→          ctx.stroke();
   274→   274→        } else if (isHovered) {
   275→   275→          ctx.strokeStyle = '#374151';
   276→   276→          ctx.lineWidth = 2;
   277→   277→          ctx.stroke();
   278→   278→        }
   279→   279→

   285→   285→          const labelText = node.label.value.length > 20

   292→   292→      const centerX = canvas.width / 2;
   293→   293→      const centerY = canvas.height / 2;
   294→   294→      const radius = Math.min(centerX, centerY) - 60;

   298→   298→        const angle = (i / filteredNodes.length) * Math.PI * 2 - Math.PI / 2;
   299→   299→        node.x = centerX + radius * Math.cos(angle);
   300→   300→        node.y = centerY + radius * Math.sin(angle);
   301→   301→      });
   302→   302→
   303→   303→      // Redraw with new positions
   304→   304→      // ... (same drawing logic as force layout)
   305→   305→    }

   319→   319→      const dx = node.x - x;

   321→   321→      return Math.sqrt(dx * dx + dy * dy) < 25;
   322→   322→    });

   324→   324→    if (clickedNode) {
   325→   325→      setSelectedNode(clickedNode.id);
   326→   326→      addActivity({
   327→   327→        type: 'query',
   328→   328→        message: createDynamicField(`Selected node: ${clickedNode.label.value}`),
   329→   329→        icon: getNodeIcon(clickedNode.type),
   330→   330→      });
   331→   331→    } else {
   332→   332→      setSelectedNode(null);
   333→   333→    }
   334→   334→  };

   345→   345→      const dx = node.x - x;
   346→   346→      const dy = node.y - y;
   347→   347→      return Math.sqrt(dx * dx + dy * dy) < 25;
   348→   348→    });
   349→   349→
   350→   350→    setHoveredNode(hovered?.id || null);

   363→   363→      },
   364→   364→      tags: ['knowledge-graph', 'visualization'],
   365→   365→    });
   366→   366→
   367→   367→    addActivity({
   368→   368→      type: 'save',
   369→   369→      message: createDynamicField(`Saved graph snapshot: ${filteredNodes.length} nodes, ${filteredEdges.length} edges`),
   370→   370→      icon: '🕸️',
   371→   371→    });
   372→   372→  };
   373→   373→
   374→   374→  return (

   376→   376→      {/* Header */}
   377→   377→      <div className="mb-6">
   378→   378→        <h1 className="text-3xl font-bold text-foreground">
   379→   379→          🕸️ {t('knowledge.title') || 'Knowledge Graph'}

   382→   382→          Explore relationships between papers, genes, compounds, and researchers
   383→   383→        </p>
   384→   384→
   385→   385→        {/* Stats */}

   389→   389→          </Badge>

   395→   395→          </Badge>
   396→   396→        </div>
   397→   397→      </div>
   398→   398→
   399→   399→      <div className="flex gap-6 h-[calc(100vh-200px)]">
   400→   400→        {/* Main Graph Canvas */}
   401→   401→        <Card className="flex-1 overflow-hidden">
