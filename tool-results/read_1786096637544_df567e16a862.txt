   860→function formatRecordCount(count: number): string {
   861→  if (count >= 1000000000) return `${(count / 1000000000).toFixed(1)}B+`;
   862→  if (count >= 1000000) return `${(count / 1000000).toFixed(0)}M+`;
   863→  if (count >= 1000) return `${(count / 1000).toFixed(0)}K+`;
   864→  return String(count);
   865→}
