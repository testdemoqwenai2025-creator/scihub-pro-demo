   565→const HighlightText = ({ text, highlight }: { text: string; highlight: string }) => {
   566→  if (!highlight.trim()) return <span>{text}</span>;
   567→  
   568→  const regex = new RegExp(`(${highlight.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
   569→  const parts = text.split(regex);
   570→  
   571→  return (
   572→    <span>
   573→      {parts.map((part, i) => 
   574→        regex.test(part) ? (
   575→          <strong key={i} className="bg-yellow-200 dark:bg-yellow-800 rounded px-0.5">{part}</strong>
   576→        ) : (
   577→          <span key={i}>{part}</span>
   578→        )
   579→      )}
   580→    </span>
   581→  );
