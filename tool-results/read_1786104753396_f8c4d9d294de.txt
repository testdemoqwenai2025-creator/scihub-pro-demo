   276→// ============ ANIMATED COUNTER HOOK ============
   277→
   278→function useAnimatedCounter(target: number, duration: number = 1500, enabled: boolean = true) {
   279→  const [count, setCount] = useState(0);
   280→  
   281→  useEffect(() => {
   282→    if (!enabled) {
   283→      setCount(target);
   284→      return;
   285→    }
   286→    
   287→    let startTime: number | null = null;
   288→    let animationFrame: number;
   289→    
   290→    const step = (timestamp: number) => {
   291→      if (!startTime) startTime = timestamp;
   292→      const progress = Math.min((timestamp - startTime) / duration, 1);
   293→      // Easing function for smooth animation
   294→      const easedProgress = 1 - Math.pow(1 - progress, 3);
   295→      setCount(Math.floor(easedProgress * target));
   296→      if (progress < 1) {
   297→        animationFrame = requestAnimationFrame(step);
   298→      }
   299→    };
   300→    
   301→    animationFrame = requestAnimationFrame(step);
   302→    return () => cancelAnimationFrame(animationFrame);
   303→  }, [target, duration, enabled]);
   304→  
   305→  return count;
   306→}

   308→// ============ SESSION DURATION TRACKER ============
   309→
   310→function useSessionDuration() {
   311→  const [sessionStart] = useState(() => {
   312→    // Check if we're in browser to avoid SSR issues
   313→    if (typeof window === 'undefined') return new Date();
   314→    try {
   315→      const stored = localStorage.getItem('scihub_session_start');
   316→      if (stored) return new Date(stored);
   317→      const now = new Date();
   318→      localStorage.setItem('scihub_session_start', now.toISOString());
   319→      return now;
   320→    } catch {
   321→      return new Date();
   322→    }
   323→  });
   324→  
   325→  const [duration, setDuration] = useState('0m');
   326→  
   327→  useEffect(() => {
   328→    const updateDuration = () => {
   329→      const diff = Date.now() - sessionStart.getTime();
   330→      const hours = Math.floor(diff / 3600000);
   331→      const minutes = Math.floor((diff % 3600000) / 60000);
   332→      if (hours > 0) {
   333→        setDuration(`${hours}h ${minutes}m`);
   334→      } else {
   335→        setDuration(`${minutes}m`);
   336→      }
   337→    };
   338→    
   339→    updateDuration();
   340→    const interval = setInterval(updateDuration, 60000);
   341→    return () => clearInterval(interval);
   342→  }, [sessionStart]);
   343→  
   344→  return duration;
   345→}
