   107→   107→// Confidence Score Badge
   108→   108→function ConfidenceBadge({ score, size = 'default' }: { score: number; size?: 'sm' | 'default' | 'lg' }) {
   109→   109→  const getColor = () => {
   110→   110→    if (score >= 90) return 'bg-green-100 text-green-800 border-green-300';
   111→   111→    if (score >= 70) return 'bg-yellow-100 text-yellow-800 border-yellow-300';
   112→   112→    return 'bg-red-100 text-red-800 border-red-300';
   113→   113→  };
   114→   114→
   115→   115→  const getLabel = () => {
   116→   116→    if (score >= 90) return 'High Confidence';
   117→   117→    if (score >= 70) return 'Medium Confidence';
   118→   118→    return 'Low Confidence';
   119→   119→  };
   120→   120→
   121→   121→  const sizeClasses = {
   122→   122→    sm: 'px-2 py-0.5 text-xs',
   123→   123→    default: 'px-3 py-1 text-sm',
   124→   124→    lg: 'px-4 py-2 text-base'
   125→   125→  };
   126→   126→
   127→   127→  return (
   128→   128→    <Badge variant="outline" className={`${getColor()} ${sizeClasses[size]}`}>
   129→   129→      {score.toFixed(1)}% • {getLabel()}
   130→   130→    </Badge>
   131→   131→  );
   132→   132→}

   152→   152→// Simplified 3D Molecule Visualization
   153→   153→function ProteinStructureViewer({ protein, confidenceScores }: ProteinVisualizationProps) {
   154→   154→  const [rotation, setRotation] = useState(0);
   155→   155→  const [showConfidence, setShowConfidence] = useState(true);
   156→   156→
   157→   157→  // Auto-rotate animation
   158→   158→  useEffect(() => {
   159→   159→    const interval = setInterval(() => {
   160→   160→      setRotation(r => (r + 0.5) % 360);
   161→   161→    }, 50);
   162→   162→    return () => clearInterval(interval);
   163→   163→  }, []);
   164→   164→
   165→   165→  // Generate simplified atom positions based on sequence length
   166→   166→  const generateAtoms = () => {
   167→   167→    const count = Math.min(protein.sequenceLength || 200, 150);
   168→   168→    const atoms = [];
   169→   169→    
   170→   170→    for (let i = 0; i < count; i++) {
   171→   171→      const angle = (i / count) * Math.PI * 4 + (rotation * Math.PI / 180);
   172→   172→      const radius = 30 + Math.sin(i * 0.1) * 20;
   173→   173→      const height = (i / count) * 120 - 60;
   174→   174→      
   175→   175→      const x = Math.cos(angle) * radius;
   176→   176→      const y = height;
   177→   177→      const z = Math.sin(angle) * radius;
   178→   178→      
   179→   179→      const confidence = confidenceScores?.[i % (confidenceScores?.length || 1)] || (protein.confidenceScore || 80);
   180→   180→      
   181→   181→      atoms.push({
   182→   182→        id: i,
   183→   183→        x, y, z,
   184→   184→        confidence,
   185→   185→        type: i % 3 === 0 ? 'alpha' : i % 3 === 1 ? 'beta' : 'coil'
   186→   186→      });
   187→   187→    }
   188→   188→    
   189→   189→    return atoms;
   190→   190→  };
   191→   191→
   192→   192→  const atoms = generateAtoms();
   193→   193→  
   194→   194→  const getAtomColor = (atom: typeof atoms[0]) => {
   195→   195→    if (!showConfidence) {
   196→   196→      return atom.type === 'alpha' ? '#3b82f6' : atom.type === 'beta' ? '#ef4444' : '#22c55e';
   197→   197→    }
   198→   198→    if (atom.confidence >= 90) return '#22c55e'; // Green for high
   199→   199→    if (atom.confidence >= 70) return '#eab308'; // Yellow for medium
   200→   200→    return '#ef4444'; // Red for low
   201→   201→  };
   202→   202→
   203→   203→  return (
   204→   204→    <div className="relative w-full h-[400px] bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 rounded-lg overflow-hidden">
   205→   205→      {/* Grid background */}
   206→   206→      <div className="absolute inset-0 opacity-10" style={{
   207→   207→        backgroundImage: `
   208→   208→          linear-gradient(rgba(255,255,255,0.1) 1px, transparent 1px),
   209→   209→          linear-gradient(90deg, rgba(255,255,255,0.1) 1px, transparent 1px)
   210→   210→        `,
   211→   211→        backgroundSize: '40px 40px'
   212→   212→      }} />
   213→   213→      
   214→   214→      {/* SVG Visualization */}
   215→   215→      <svg viewBox="-100 -80 200 160" className="w-full h-full" style={{ transform: 'rotateX(15deg)' }}>
   216→   216→        {/* Bonds between nearby atoms */}
   217→   217→        {atoms.map((atom, i) => 
   218→   218→          i > 0 && (
   219→   219→            <line
   220→   220→              key={`bond-${i}`}
   221→   221→              x1={atoms[i-1].x}
   222→   222→              y1={atoms[i-1].y}
   223→   223→              x2={atom.x}
   224→   224→              y2={atom.y}
   225→   225→              stroke="rgba(255,255,255,0.2)"
   226→   226→              strokeWidth="0.5"
   227→   227→            />
   228→   228→          )
   229→   229→        )}
   230→   230→        
   231→   231→        {/* Atoms */}
   232→   232→        {atoms.map((atom) => (
   233→   233→          <circle
   234→   234→            key={atom.id}
   235→   235→            cx={atom.x}
   236→   236→            cy={atom.y}
   237→   237→            r={atom.type === 'alpha' ? 2.5 : atom.type === 'beta' ? 2 : 1.5}
   238→   238→            fill={getAtomColor(atom)}
   239→   239→            opacity={0.9}
   240→   240→          />
   241→   241→        ))}
   242→   242→        
   243→   243→        {/* Center label */}
   244→   244→        <text x="0" y="75" textAnchor="middle" fill="rgba(255,255,255,0.6)" fontSize="6" fontFamily="monospace">
   245→   245→          {protein.uniprotId}
   246→   246→        </text>
   247→   247→      </svg>
   248→   248→
   249→   249→      {/* Controls overlay */}
   250→   250→      <div className="absolute bottom-4 left-4 right-4 flex justify-between items-center">
   251→   251→        <div className="flex gap-2">
   252→   252→          <Button
   253→   253→            variant="secondary"
   254→   254→            size="sm"
   255→   255→            onClick={() => setShowConfidence(!showConfidence)}
   256→   256→            className="bg-black/50 hover:bg-black/70 text-white border-white/20"
   257→   257→          >
   258→   258→            {showConfidence ? 'Show Type' : 'Show Confidence'}
   259→   259→          </Button>
   260→   260→        </div>
   261→   261→        
   262→   262→        <Badge variant="outline" className="bg-black/50 text-white border-white/20">
   263→   263→          {atoms.length} residues visualized
   264→   264→        </Badge>
   265→   265→      </div>
   266→   266→
   267→   267→      {/* Legend */}
   268→   268→      <div className="absolute top-4 right-4 bg-black/50 rounded-lg p-3 backdrop-blur-sm">
   269→   269→        <p className="text-xs text-white font-semibold mb-2">Legend</p>
   270→   270→        {showConfidence ? (
   271→   271→          <>
   272→   272→            <div className="flex items-center gap-2 mb-1">
   273→   273→              <div className="w-3 h-3 rounded-full bg-green-500" />
   274→   274→              <span className="text-xs text-gray-300">High (&ge;90%)</span>
   275→   275→            </div>
   276→   276→            <div className="flex items-center gap-2 mb-1">
   277→   277→              <div className="w-3 h-3 rounded-full bg-yellow-500" />
   278→   278→              <span className="text-xs text-gray-300">Medium (70-90%)</span>
   279→   279→            </div>
   280→   280→            <div className="flex items-center gap-2">
   281→   281→              <div className="w-3 h-3 rounded-full bg-red-500" />
   282→   282→              <span className="text-xs text-gray-300">Low (&lt;70%)</span>
   283→   283→            </div>
   284→   284→          </>
   285→   285→        ) : (
   286→   286→          <>
   287→   287→            <div className="flex items-center gap-2 mb-1">
   288→   288→              <div className="w-3 h-3 rounded-full bg-blue-500" />
   289→   289→              <span className="text-xs text-gray-300">Alpha helix</span>
   290→   290→            </div>
   291→   291→            <div className="flex items-center gap-2 mb-1">
   292→   292→              <div className="w-3 h-3 rounded-full bg-red-500" />
   293→   293→              <span className="text-xs text-gray-300">Beta sheet</span>
   294→   294→            </div>
   295→   295→            <div className="flex items-center gap-2">
   296→   296→              <div className="w-3 h-3 rounded-full bg-green-500" />
   297→   297→              <span className="text-xs text-gray-300">Coil</span>
   298→   298→            </div>
   299→   299→          </>
   300→   300→        )}
   301→   301→      </div>
   302→   302→    </div>
   303→   303→  );
   304→   304→}

   420→   420→// Main AlphaFold Component
   421→   421→export function AlphaFoldConnector() {
   422→   422→  // State management
   423→   423→  const [activeTab, setActiveTab] = useState('search');
   424→   424→  const [searchQuery, setSearchQuery] = useState('');
   425→   425→  const [isSearching, setIsSearching] = useState(false);
   426→   426→  const [searchResults, setSearchResults] = useState<AlphaFoldSearchResult | null>(null);
   427→   427→  const [selectedProtein, setSelectedProtein] = useState<AlphaFoldProtein | null>(null);
   428→   428→  const [structureData, setStructureData] = useState<any>(null);
   429→   429→  const [isLoadingStructure, setIsLoadingStructure] = useState(false);
   430→   430→  const [featuredProteins, setFeaturedProteins] = useState<AlphaFoldProtein[]>([]);
   431→   431→  const [dbStats, setDbStats] = useState<any>(null);
   432→   432→  const [batchInput, setBatchInput] = useState('');
   433→   433→  const [batchResults, setBatchResults] = useState<BatchQueryResult | null>(null);
   434→   434→  const [isBatchProcessing, setIsBatchProcessing] = useState(false);
   435→   435→
   436→   436→  // Load featured proteins and stats on mount
   437→   437→  useEffect(() => {
   438→   438→    loadInitialData();
   439→   439→  }, []);
   440→   440→
   441→   441→  const loadInitialData = async () => {
   442→   442→    try {
   443→   443→      const [proteins, stats] = await Promise.all([
   444→   444→        getFeaturedProteins(),
   445→   445→        getDatabaseStats()
   446→   446→      ]);
   447→   447→      setFeaturedProteins(proteins);
   448→   448→      setDbStats(stats);
   449→   449→    } catch (error) {
   450→   450→      console.error('Error loading initial data:', error);
   451→   451→    }
   452→   452→  };
   453→   453→
   454→   454→  // Search handler
   455→   455→  const handleSearch = async () => {
   456→   456→    if (!searchQuery.trim()) return;
   457→   457→    
   458→   458→    setIsSearching(true);
   459→   459→    try {
   460→   460→      // Determine if it's a UniProt ID (starts with letter followed by numbers/letters)
   461→   461→      const isUniProtId = /^[A-Z][0-9A-Z]{1,10}$/i.test(searchQuery.trim());
   462→   462→      
   463→   463→      let results: AlphaFoldSearchResult;
   464→   464→      
   465→   465→      if (isUniProtId) {
   466→   466→        // Direct UniProt ID lookup
   467→   467→        const protein = await fetchProteinByUniProt(searchQuery.trim());
   468→   468→        results = {
   469→   469→          query: searchQuery.trim(),
   470→   470→          results: [protein],
   471→   471→          totalFound: 1,
   472→   472→          searchTime: Date.now(),
   473→   473→          source: 'alphafold'
   474→   474→        };
   475→   475→      } else {
   476→   476→        // Keyword search
   477→   477→        results = await searchProteins(searchQuery.trim());
   478→   478→      }
   479→   479→      
   480→   480→      setSearchResults(results);
   481→   481→    } catch (error) {
   482→   482→      console.error('Search error:', error);
   483→   483→    } finally {
   484→   484→      setIsSearching(false);
   485→   485→    }
   486→   486→  };
