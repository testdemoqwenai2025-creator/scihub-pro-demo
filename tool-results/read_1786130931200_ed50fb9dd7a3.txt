   665→function createVolcanoPlot(containerId) {
   666→  const container = document.getElementById(containerId);
   667→  
   668→  // Setup SVG
   669→  const width = 800;
   670→  const height = 500;
   671→  const margin = { top: 40, right: 40, bottom: 60, left: 70 };
   672→  
   673→  // Scales
   674→  const xScale = d3.scaleLinear()
   675→    .domain(d3.extent(data.genes, d => d.foldChange))
   676→    .range([margin.left, width - margin.right]);
   677→    
   678→  const yScale = d3.scaleLinear()
   679→    .domain([0, -Math.log10(d3.min(data.genes, d => d.pValue)) * 1.1])
   680→    .range([height - margin.bottom, margin.top]);
   681→  
   682→  // Draw points
   683→  const svg = d3.select(container).append('svg')
   684→    .attr('width', width)
   685→    .attr('height', height);
   686→    
   687→  svg.selectAll('circle')
   688→    .data(data.genes)
   689→    .enter()
   690→    .append('circle')
   691→    .attr('cx', d => xScale(d.foldChange))
   692→    .attr('cy', d => yScale(-Math.log10(d.pValue)))
   693→    .attr('r', d => d.significant ? 6 : 4)
   694→    .attr('fill', d => {
   695→      if (d.significant && d.foldChange > 0) return '#e41a1c'; // Red up
   696→      if (d.significant && d.foldChange < 0) return '#377eb8'; // Blue down
   697→      return '#999999'; // Gray not significant
   698→    })
   699→    .attr('opacity', 0.7)
   700→    .on('mouseover', showTooltip)
   701→    .on('mouseout', hideTooltip);
   702→    
   703→  // Threshold lines
   704→  svg.append('line')
   705→    .attr('x1', xScale(-1))
   706→    .attr('y1', margin.top)
   707→    .attr('x2', xScale(-1))
   708→    .attr('y2', height - margin.bottom)
   709→    .attr('stroke', 'green')
   710→    .attr('stroke-dasharray', '5,5');
   711→    
   712→  svg.append('line')
   713→    .attr('x1', xScale(1))
   714→    .attr('y1', margin.top)
   715→    .attr('x2', xScale(1))
   716→    .attr('y2', height - margin.bottom)
   717→    .attr('stroke', 'green')
   718→    .attr('stroke-dasharray', '5,5');
   719→    
   720→  svg.append('line')
   721→    .attr('x1', margin.left)
   722→    .attr('y1', yScale(-Math.log10(0.05)))
   723→    .attr('x2', width - margin.right)
   724→    .attr('y2', yScale(-Math.log10(0.05)))
   725→    .attr('stroke', 'blue')
   726→    .attr('stroke-dasharray', '5,5');
   727→    
   728→  // Axes
   729→  svg.append('g')
   730→    .attr('transform', \`translate(0, \${height - margin.bottom})\`)
   731→    .call(d3.axisBottom(xScale));
   732→    
   733→  svg.append('g')
   734→    .attr('transform', \`translate(\${margin.left}, 0)\`)
   735→    .call(d3.axisLeft(yScale));
   736→    
   737→  // Labels
   738→  svg.append('text')
   739→    .attr('x', width / 2)
   740→    .attr('y', height - 10)

  1191→const generateMarkdownOutput = (code: string): string => {
  1192→  const outputs: string[] = [];
  1193→  
  1194→  outputs.push('');
  1195→  outputs.push('#'.repeat(60));
  1196→  outputs.push('# RENDERED MARKDOWN PREVIEW');

  1202→  let inCodeBlock = false;
  1203→  
  1204→  for (const line of lines) {

  1206→      inCodeBlock = !inCodeBlock;
  1207→      if (inCodeBlock) {
  1208→        outputs.push('<div class="code-block">');
  1209→      } else {
  1210→        outputs.push('</div>');
  1211→      }
  1212→      continue;
  1213→    }
  1214→    
  1215→    if (inCodeBlock) {
  1216→      outputs.push('  ' + line);
  1217→      continue;
  1218→    }
  1219→    
  1220→    // Headers
  1221→    if (line.startsWith('# ')) {
  1222→      outputs.push(`<h1>${line.substring(2)}</h1>`);
  1223→    } else if (line.startsWith('## ')) {

  1226→      outputs.push(`<h3>${line.substring(4)}</h3>`);
  1227→    } else if (line.startsWith('> ')) {
  1228→      outputs.push(`<blockquote>${line.substring(2)}</blockquote>`);
  1229→    } else if (line.startsWith('- ') || line.startsWith('* ')) {
  1230→      outputs.push(`<li>${line.substring(2)}</li>`);
  1231→    } else if (line.match(/^\d+\.\s/)) {
  1232→      outputs.push(`<li>${line.replace(/^\d+\.\s/, '')}</li>`);
  1233→    } else if (line.trim() === '---') {
  1234→      outputs.push('<hr/>');
  1235→    } else if (line.trim()) {
  1236→      outputs.push(`<p>${line}</p>`);
  1237→    }
  1238→  }
  1239→  
  1240→  outputs.push('');
  1241→  outputs.push('#'.repeat(60));
  1242→  outputs.push('*Full Markdown rendering would include syntax highlighting,');
  1243→  outputs.push('table formatting, image embedding, and MathJax support*');
  1244→  outputs.push('#'.repeat(60));
