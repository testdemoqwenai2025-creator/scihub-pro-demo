   190→// ============ LAYOUT ALGORITHMS ============
   191→
   192→function applyForceLayout(nodes: GraphNode[], edges: GraphEdge[]): GraphNode[] {
   193→  const centerX = SVG_WIDTH / 2;
   194→  const centerY = SVG_HEIGHT / 2;
   195→  const iterations = 100;
   196→  
   197→  // Initialize positions if not set
   198→  const positionedNodes = nodes.map(node => ({
   199→    ...node,
   200→    x: node.x || centerX + (Math.random() - 0.5) * 200,
   201→    y: node.y || centerY + (Math.random() - 0.5) * 200,
   202→  }));
   203→  
   204→  // Create adjacency map for faster lookup
   205→  const adjacencyMap = new Map<string, { node: GraphNode; edge: GraphEdge}[]>();
   206→  edges.forEach(edge => {
   207→    if (!adjacencyMap.has(edge.source)) adjacencyMap.set(edge.source, []);
   208→    if (!adjacencyMap.has(edge.target)) adjacencyMap.set(edge.target, []);
   209→    const sourceNode = positionedNodes.find(n => n.id === edge.source);
   210→    const targetNode = positionedNodes.find(n => n.id === edge.target);
   211→    if (sourceNode) adjacencyMap.get(edge.source)!.push({ node: targetNode!, edge });
   212→    if (targetNode) adjacencyMap.get(edge.target)!.push({ node: sourceNode!, edge });
   213→  });
   214→  
   215→  // Simulate force-directed layout
   216→  for (let iter = 0; iter < iterations; iter++) {
   217→    const temperature = 1 - iter / iterations;
   218→    
   219→    positionedNodes.forEach((node, i) => {
   220→      let fx = 0, fy = 0;
   221→      
   222→      // Repulsion between all nodes
   223→      positionedNodes.forEach((other, j) => {
   224→        if (i === j) return;
   225→        const dx = node.x - other.x;
   226→        const dy = node.y - other.y;
   227→        const dist = Math.sqrt(dx * dx + dy * dy) || 1;
   228→        const force = 1500 / (dist * dist);
   229→        fx += (dx / dist) * force;
   230→        fy += (dy / dist) * force;
   231→      });
   232→      
   233→      // Attraction along edges
   234→      const neighbors = adjacencyMap.get(node.id) || [];
   235→      neighbors.forEach(({ node: other, edge }) => {
   236→        const dx = other.x - node.x;
   237→        const dy = other.y - node.y;
   238→        const dist = Math.sqrt(dx * dx + dy * dy) || 1;
   239→        const force = dist * 0.01 * edge.weight;
   240→        fx += (dx / dist) * force;
   241→        fy += (dy / dist) * force;
   242→      });
   243→      
   244→      // Gravity toward center
   245→      fx += (centerX - node.x) * 0.01;
   246→      fy += (centerY - node.y) * 0.01;
   247→      
   248→      // Apply forces with temperature damping
   249→      node.x += fx * temperature * 0.1;
   250→      node.y += fy * temperature * 0.1;
   251→      
   252→      // Keep within bounds
   253→      node.x = Math.max(50, Math.min(SVG_WIDTH - 50, node.x));
   254→      node.y = Math.max(50, Math.min(SVG_HEIGHT - 50, node.y));
   255→    });
   256→  }
   257→  
   258→  return positionedNodes;
   259→}

   261→function applyCircularLayout(nodes: GraphNode[]): GraphNode[] {
   262→  const centerX = SVG_WIDTH / 2;
   263→  const centerY = SVG_HEIGHT / 2;
   264→  const radius = Math.min(centerX, centerY) - 80;
   265→  
   266→  return nodes.map((node, i) => ({
   267→    ...node,
   268→    x: centerX + radius * Math.cos((i / nodes.length) * Math.PI * 2 - Math.PI / 2),
   269→    y: centerY + radius * Math.sin((i / nodes.length) * Math.PI * 2 - Math.PI / 2),
   270→  }));
   271→}
   272→
   273→function applyHierarchicalLayout(nodes: GraphNode[], edges: GraphEdge[]): GraphNode[] {
   274→  // Group nodes by type for hierarchical arrangement
   275→  const typeOrder: NodeType[] = ['concept', 'paper', 'gene', 'compound', 'author', 'institution'];
   276→  const groups: Record<string, GraphNode[]> = {};
   277→  
   278→  typeOrder.forEach(type => {
   279→    groups[type] = nodes.filter(n => n.type === type);
   280→  });
   281→  
   282→  const levelHeight = (SVG_HEIGHT - 100) / typeOrder.length;
   283→  const positionedNodes: GraphNode[] = [];
   284→  
   285→  typeOrder.forEach((type, levelIndex) => {
   286→    const groupNodes = groups[type];
   287→    const y = 80 + levelIndex * levelHeight + levelHeight / 2;
   288→    const groupWidth = (SVG_WIDTH - 100) / (groupNodes.length + 1);
   289→    
   290→    groupNodes.forEach((node, i) => {
   291→      positionedNodes.push({
   292→        ...node,
   293→        x: 50 + groupWidth * (i + 1),
   294→        y,
   295→      });
   296→    });
   297→  });
   298→  
   299→  return positionedNodes;
   300→}
   301→
   302→function applyGridLayout(nodes: GraphNode[]): GraphNode[] {
   303→  const cols = Math.ceil(Math.sqrt(nodes.length));
   304→  const rows = Math.ceil(nodes.length / cols);
   305→  const cellWidth = (SVG_WIDTH - 100) / cols;
   306→  const cellHeight = (SVG_HEIGHT - 100) / rows;
   307→  
   308→  return nodes.map((node, i) => ({
   309→    ...node,
   310→    x: 50 + cellWidth * (i % cols) + cellWidth / 2,
   311→    y: 50 + cellHeight * Math.floor(i / cols) + cellHeight / 2,
   312→  }));
   313→}

   333→function getNodeRadius(node: GraphNode, sizeBy: GraphViewSettings['nodeSizeBy']): number {
   334→  switch (sizeBy) {
   335→    case 'connections':
   336→      return 15 + Math.min(node.connections, 10) * 2.5;
