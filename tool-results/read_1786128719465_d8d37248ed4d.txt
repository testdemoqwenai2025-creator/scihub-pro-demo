   277→// ============ ANIMATED COUNTER HOOK ============
   278→
   279→function useAnimatedCounter(target: number, duration: number = 1500, enabled: boolean = true) {
   280→  const [count, setCount] = useState(0);
   281→  
   282→  useEffect(() => {
   283→    if (!enabled) {
   284→      setCount(target);
   285→      return;
   286→    }
   287→    
   288→    let startTime: number | null = null;
   289→    let animationFrame: number;
   290→    
   291→    const step = (timestamp: number) => {
   292→      if (!startTime) startTime = timestamp;
   293→      const progress = Math.min((timestamp - startTime) / duration, 1);
   294→      // Easing function for smooth animation
   295→      const easedProgress = 1 - Math.pow(1 - progress, 3);
   296→      setCount(Math.floor(easedProgress * target));
   297→      if (progress < 1) {
   298→        animationFrame = requestAnimationFrame(step);
   299→      }
   300→    };
   301→    
   302→    animationFrame = requestAnimationFrame(step);
   303→    return () => cancelAnimationFrame(animationFrame);
   304→  }, [target, duration, enabled]);
   305→  
   306→  return count;
   307→}

   309→// ============ SESSION DURATION TRACKER ============
   310→
   311→function useSessionDuration() {
   312→  const [sessionStart] = useState(() => {
   313→    // Check if we're in browser to avoid SSR issues
   314→    if (typeof window === 'undefined') return new Date();
   315→    try {
   316→      const stored = localStorage.getItem('scihub_session_start');
   317→      if (stored) return new Date(stored);
   318→      const now = new Date();
   319→      localStorage.setItem('scihub_session_start', now.toISOString());
   320→      return now;
   321→    } catch {
   322→      return new Date();
   323→    }
   324→  });
   325→  
   326→  const [duration, setDuration] = useState('0m');
   327→  
   328→  useEffect(() => {
   329→    const updateDuration = () => {
   330→      const diff = Date.now() - sessionStart.getTime();
   331→      const hours = Math.floor(diff / 3600000);
   332→      const minutes = Math.floor((diff % 3600000) / 60000);
   333→      if (hours > 0) {
   334→        setDuration(`${hours}h ${minutes}m`);
   335→      } else {
   336→        setDuration(`${minutes}m`);
   337→      }
   338→    };
   339→    
   340→    updateDuration();
   341→    const interval = setInterval(updateDuration, 60000);
   342→    return () => clearInterval(interval);
   343→  }, [sessionStart]);
   344→  
   345→  return duration;
   346→}
