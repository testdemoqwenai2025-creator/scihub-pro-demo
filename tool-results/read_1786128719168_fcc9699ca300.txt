     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useTranslation } from '@/i18n/useTranslation';
     5→import { useDynamicStore, createDynamicField } from '@/store/useDynamicStore';
     6→import { executeSimulatedQuery, resultToCSV, resultToJSON, getFreeTierInfo } from '@/services/databaseSimulation';
     7→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
     8→import { Badge } from '@/components/ui/badge';
     9→import { Button } from '@/components/ui/button';
    10→import { Input } from '@/components/ui/input';
    11→import { Progress } from '@/components/ui/progress';
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from '@/components/ui/select';
    19→import { DataLakeSkeleton, CardSkeleton, StatsCardSkeleton } from '@/components/SkeletonComponents';
    20→import { showSuccessToast, showErrorToast, showLoadingToast, showInfoToast } from '@/lib/toast-utils';
    21→
    22→// ============ TYPES ============
    23→
    24→interface DatasetFormData {
    25→  name: string;
    26→  description: string;
    27→  type: string;
    28→  format: string;
    29→  sourceUrl: string;
    30→  tags: string[];
    31→  isPublic: boolean;
    32→}
    33→
    34→// ============ DATA LAKE PAGE ============
    35→
    36→export default function DataLakePage() {
    37→  const { t } = useTranslation();
    38→  const {
    39→    datasets,
    40→    createDataset,
    41→    updateDataset,
    42→    deleteDataset,
    43→    getTotalStorage,
    44→    addActivity,
    45→    checkVolumeThreshold,
    46→    dbConfig,
    47→  } = useDynamicStore();
    48→
    49→  // UI State
    50→  const [searchQuery, setSearchQuery] = useState('');
    51→  const [viewMode, setViewMode] = useState<'grid' | 'table'>('grid');
    52→  const [selectedDataset, setSelectedDataset] = useState<string | null>(null);
    53→  const [showCreateForm, setShowCreateForm] = useState(false);
    54→  const [isUploading, setIsUploading] = useState(false);
    55→  const [uploadProgress, setUploadProgress] = useState(0);
    56→  const [editingField, setEditingField] = useState<{ id: string; field: string } | null>(null);
    57→  const [tempValue, setTempValue] = useState('');
    58→  
    59→  // Form state
    60→  const [formData, setFormData] = useState<DatasetFormData>({
    61→    name: '',
    62→    description: '',
    63→    type: 'Genomics',
    64→    format: 'CSV',
    65→    sourceUrl: '',
    66→    tags: [],
    67→    isPublic: true,
    68→  });
    69→
    70→  // Volume threshold
    71→  const [volumeStatus, setVolumeStatus] = useState(checkVolumeThreshold());
    72→  const [isLoading, setIsLoading] = useState(true);
    73→
    74→  // Simulate loading on mount
    75→  useEffect(() => {
    76→    const timer = setTimeout(() => setIsLoading(false), 1500);
    77→    return () => clearTimeout(timer);
    78→  }, []);
    79→
    80→  // Check volume periodically
    81→  useEffect(() => {
    82→    const interval = setInterval(() => {
    83→      setVolumeStatus(checkVolumeThreshold());
    84→    }, 10000);
    85→    return () => clearInterval(interval);
    86→  }, [checkVolumeThreshold]);
    87→
    88→  // Filter datasets
    89→  const filteredDatasets = datasets.filter(dataset => {
    90→    const matchesSearch = 
    91→      dataset.name.value.toLowerCase().includes(searchQuery.toLowerCase()) ||
    92→      dataset.description.value.toLowerCase().includes(searchQuery.toLowerCase()) ||
    93→      dataset.tags.value.some(tag => tag.toLowerCase().includes(searchQuery.toLowerCase()));
    94→    return matchesSearch;
    95→  });
    96→
    97→  // Storage calculations
    98→  const totalStorageMB = (getTotalStorage() / 1024 / 1024).toFixed(2);
    99→  const totalRows = datasets.reduce((acc, ds) => acc + ds.rows.value, 0);
   100→  const publicCount = datasets.filter(ds => ds.isPublic.value).length;
   101→
   102→  // Handle dataset creation
   103→  const handleCreateDataset = async () => {
   104→    if (!formData.name.trim()) return;
   105→
   106→    setIsUploading(true);
   107→    setUploadProgress(0);
   108→    showLoadingToast('Creating Dataset...', `Uploading "${formData.name}"`);
   109→
   110→    // Simulate upload progress
   111→    for (let i = 0; i <= 100; i += 10) {
   112→      await new Promise(resolve => setTimeout(resolve, 100));
   113→      setUploadProgress(i);
   114→    }
   115→
   116→    const datasetSize = (Math.random() * 500 + 50).toFixed(1);
   117→    
   118→    createDataset({
   119→      name: createDynamicField(formData.name),
   120→      description: createDynamicField(formData.description || `Dataset: ${formData.name}`),
   121→      size: createDynamicField(`${datasetSize} MB`),
   122→      rows: createDynamicField(Math.floor(Math.random() * 1000000 + 10000)),
   123→      columns: createDynamicField(Math.floor(Math.random() * 100 + 10)),
   124→      type: createDynamicField(formData.type),
   125→      format: createDynamicField(formData.format),
   126→      sourceUrl: createDynamicField(formData.sourceUrl || '#'),
   127→      tags: createDynamicField(formData.tags.length > 0 ? formData.tags : ['user-uploaded']),
   128→      isPublic: createDynamicField(formData.isPublic),
   129→      storageLocation: volumeStatus.exceeded ? 'duckdb' : 'local',
   130→    });
   131→
   132→    setIsUploading(false);
   133→    setShowCreateForm(false);
   134→    setFormData({
   135→      name: '',
   136→      description: '',
   137→      type: 'Genomics',
   138→      format: 'CSV',
   139→      sourceUrl: '',
   140→      tags: [],
   141→      isPublic: true,
   142→    });
   143→
   144→    addActivity({
   145→      type: 'upload',
   146→      message: createDynamicField(`Uploaded dataset "${formData.name}"`),
   147→      icon: '📊',
   148→    });
   149→    
   150→    showSuccessToast('Dataset Created!', `Upload completed! File size: ${datasetSize} MB`);
   151→  };
   152→
   153→  // Handle dataset deletion
   154→  const handleDeleteDataset = (id: string) => {
   155→    const dataset = datasets.find(d => d.id === id);
   156→    if (dataset && confirm(`Delete "${dataset.name.value}"? This action cannot be undone.`)) {
   157→      deleteDataset(id);
   158→      setSelectedDataset(null);
   159→      showInfoToast('Dataset Deleted', `"${dataset.name.value}" has been removed`);
   160→    }
   161→  };
   162→
   163→  // Handle inline editing
   164→  const startEditing = (id: string, field: string, currentValue: any) => {
   165→    setEditingField({ id, field });
   166→    setTempValue(String(currentValue));
   167→  };
   168→
   169→  const saveEdit = (id: string, field: string) => {
   170→    const dataset = datasets.find(d => d.id === id);
   171→    if (dataset) {
   172→      const currentValue = dataset[field as keyof typeof dataset];
   173→      updateDataset(id, {
   174→        [field]: {
   175→          // Handle DynamicField or plain value
   176→          ...(typeof currentValue === 'object' && currentValue !== null ? currentValue : { value: currentValue }),
   177→          value: tempValue,
   178→          isDirty: true,
   179→          lastModified: new Date(),
   180→        }
   181→      });
   182→    }
   183→    setEditingField(null);
   184→  };
   185→
   186→  // Query dataset with DuckDB simulation
   187→  const handleQueryDataset = async (datasetId: string) => {
   188→    const dataset = datasets.find(d => d.id === datasetId);
   189→    if (!dataset) return;
   190→
   191→    try {
   192→      const sql = `SELECT * FROM ${dataset.name.value.replace(/\s+/g, '_').toLowerCase()} LIMIT 20`;
   193→      const result = await executeSimulatedQuery(sql, dataset.storageLocation);
   194→      
   195→      addActivity({
   196→        type: 'query',
   197→        message: createDynamicField(`Queried ${dataset.name.value}: ${result.rows.length} rows returned`),
   198→        icon: '🔍',
   199→      });
   200→
   201→      // Could open results in a modal or navigate to query page
   202→      console.log('Query Result:', result);
   203→    } catch (error) {
   204→      console.error('Query failed:', error);
   205→    }
   206→  };
   207→
   208→  // Export dataset
   209→  const handleExportDataset = async (datasetId: string, format: 'csv' | 'json') => {
   210→    const dataset = datasets.find(d => d.id === datasetId);
   211→    if (!dataset) return;
   212→
   213→    // Generate mock export data
   214→    const mockResult = await executeSimulatedQuery(
   215→      `SELECT * FROM ${dataset.name.value.replace(/\s+/g, '_').toLowerCase()} LIMIT 100`
   216→    );
   217→
   218→    const content = format === 'csv' ? resultToCSV(mockResult) : resultToJSON(mockResult);
   219→    const blob = new Blob([content], { type: format === 'csv' ? 'text/csv' : 'application/json' });
   220→    const url = URL.createObjectURL(blob);
   221→    const a = document.createElement('a');
   222→    a.href = url;
   223→    a.download = `${dataset.name.value}.${format}`;
   224→    a.click();
   225→    URL.revokeObjectURL(url);
   226→
   227→    addActivity({
   228→      type: 'export',
   229→      message: createDynamicField(`Exported ${dataset.name.value} as ${format.toUpperCase()}`),
   230→      icon: '📥',
   231→    });
   232→    
   233→    showSuccessToast('Export Complete', `${dataset.name.value} exported as ${format.toUpperCase()}`);
   234→  };
   235→
   236→  // Loading skeleton
   237→  if (isLoading) {
   238→    return (
   239→      <div className="min-h-screen bg-background">
   240→        <DataLakeSkeleton />
   241→      </div>
   242→    );
   243→  }
   244→
   245→  return (
   246→    <div className="min-h-screen bg-background p-6">
   247→      {/* Header */}
   248→      <div className="mb-8">
   249→        <h1 className="text-3xl font-bold text-foreground">{t('data.title') || 'Data Lake Browser'}</h1>
   250→        <p className="text-muted-foreground mt-1">
   251→          Browse, manage, and analyze scientific datasets. All data stored locally or pushed to free-tier databases.
   252→        </p>
   253→        
   254→        {/* Free Tier Info */}
   255→        <div className="mt-3 flex items-center gap-3 text-sm">
   256→          <Badge variant="secondary" className="bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200">
   257→            🆓 Free Tier Active
   258→          </Badge>
   259→          <span className="text-muted-foreground">
   260→            DuckDB • IndexedDB • localStorage
   261→          </span>
   262→          {volumeStatus.shouldPush && (
   263→            <Badge variant="outline" className="border-yellow-400 text-yellow-600 animate-pulse">
   264→              ⚠️ Approaching storage limit
   265→            </Badge>
   266→          )}
   267→        </div>
   268→      </div>
   269→
   270→      {/* Controls */}
   271→      <div className="flex flex-col md:flex-row gap-4 mb-6">
   272→        <div className="flex-1 flex gap-3">
   273→          <Input
   274→            placeholder="Search datasets by name, description, or tags..."
   275→            value={searchQuery}
   276→            onChange={(e) => setSearchQuery(e.target.value)}
   277→            className="max-w-md"
   278→          />
   279→          
   280→          <Select value={viewMode} onValueChange={(v) => setViewMode(v as 'grid' | 'table')}>
   281→            <SelectTrigger className="w-[120px]">
   282→              <SelectValue />
   283→            </SelectTrigger>
   284→            <SelectContent>
   285→              <SelectItem value="grid">Grid View</SelectItem>
   286→              <SelectItem value="table">Table View</SelectItem>
   287→            </SelectContent>
   288→          </Select>
   289→        </div>
   290→
   291→        <Button onClick={() => setShowCreateForm(!showCreateForm)}>
   292→          + Upload Dataset
   293→        </Button>
   294→      </div>
   295→
   296→      {/* Stats Bar */}
   297→      <div className="grid grid-cols-2 md:grid-cols-5 gap-4 mb-8">
   298→        <Card>
   299→          <CardContent className="p-4 flex items-center gap-3">
   300→            <span className="text-2xl">📊</span>
   301→            <div>
   302→              <p className="text-xs text-muted-foreground">Total Datasets</p>
   303→              <p className="text-xl font-bold">{datasets.length}</p>
   304→            </div>
   305→          </CardContent>
   306→        </Card>
   307→        <Card>
   308→          <CardContent className="p-4 flex items-center gap-3">
   309→            <span className="text-2xl">💾</span>
   310→            <div>
   311→              <p className="text-xs text-muted-foreground">Total Size</p>
   312→              <p className="text-xl font-bold">{totalStorageMB} MB</p>
   313→            </div>
   314→          </CardContent>
   315→        </Card>
   316→        <Card>
   317→          <CardContent className="p-4 flex items-center gap-3">
   318→            <span className="text-2xl">📋</span>
   319→            <div>
   320→              <p className="text-xs text-muted-foreground">Total Rows</p>
   321→              <p className="text-xl font-bold">{totalRows.toLocaleString()}</p>
   322→            </div>
   323→          </CardContent>
   324→        </Card>
   325→        <Card>
   326→          <CardContent className="p-4 flex items-center gap-3">
   327→            <span className="text-2xl">🌐</span>
   328→            <div>
   329→              <p className="text-xs text-muted-foreground">Public</p>
   330→              <p className="text-xl font-bold text-green-500">{publicCount}</p>
   331→            </div>
   332→          </CardContent>
   333→        </Card>
   334→        <Card>
   335→          <CardContent className="p-4 flex items-center gap-3">
   336→            <span className="text-2xl">{dbConfig.provider === 'duckdb' ? '🦆' : '🐘'}</span>
   337→            <div>
   338→              <p className="text-xs text-muted-foreground">Storage Engine</p>
   339→              <p className="text-lg font-bold capitalize">{dbConfig.provider}</p>
   340→            </div>
   341→          </CardContent>
   342→        </Card>
   343→      </div>
   344→
   345→      {/* Create Dataset Form */}
   346→      {showCreateForm && (
   347→        <Card className="mb-8 border-primary">
   348→          <CardHeader>
   349→            <CardTitle>Upload New Dataset</CardTitle>
   350→            <CardDescription>
   351→              Fill in the details below. Your data will be stored using the configured database provider.
   352→            </CardDescription>
   353→          </CardHeader>
   354→          <CardContent>
   355→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-4">
   356→              <div>
   357→                <label className="text-sm font-medium">Dataset Name *</label>
   358→                <Input
   359→                  value={formData.name}
   360→                  onChange={(e) => setFormData({...formData, name: e.target.value})}
   361→                  placeholder="e.g., TCGA Expression Matrix"
   362→                  className="mt-1"
   363→                />
   364→              </div>
   365→              <div>
   366→                <label className="text-sm font-medium">Data Type</label>
   367→                <Select value={formData.type} onValueChange={(v) => setFormData({...formData, type: v})}>
   368→                  <SelectTrigger className="mt-1">
   369→                    <SelectValue />
   370→                  </SelectTrigger>
   371→                  <SelectContent>
   372→                    <SelectItem value="Genomics">Genomics</SelectItem>
   373→                    <SelectItem value="Proteomics">Proteomics</SelectItem>
   374→                    <SelectItem value="Chemical Biology">Chemical Biology</SelectItem>
   375→                    <SelectItem value="Structural Biology">Structural Biology</SelectItem>
   376→                    <SelectItem value="Literature">Literature</SelectItem>
   377→                    <SelectItem value="Clinical">Clinical Data</SelectItem>
   378→                    <SelectItem value="Imaging">Imaging</SelectItem>
   379→                    <SelectItem value="Other">Other</SelectItem>
   380→                  </SelectContent>
   381→                </Select>
   382→              </div>
   383→              <div className="md:col-span-2">
   384→                <label className="text-sm font-medium">Description</label>
   385→                <Input
   386→                  value={formData.description}
   387→                  onChange={(e) => setFormData({...formData, description: e.target.value})}
   388→                  placeholder="Brief description of the dataset..."
   389→                  className="mt-1"
   390→                />
   391→              </div>
   392→              <div>
   393→                <label className="text-sm font-medium">Format</label>
   394→                <Select value={formData.format} onValueChange={(v) => setFormData({...formData, format: v})}>
   395→                  <SelectTrigger className="mt-1">
   396→                    <SelectValue />
   397→                  </SelectTrigger>
   398→                  <SelectContent>
   399→                    <SelectItem value="CSV">CSV</SelectItem>
   400→                    <SelectItem value="JSON">JSON</SelectItem>
   401→                    <SelectItem value="Parquet">Parquet</SelectItem>
   402→                    <SelectItem value="HDF5">HDF5</SelectItem>
   403→                    <SelectItem value="VCF">VCF/BCF</SelectItem>
   404→                    <SelectItem value="FASTA">FASTA</SelectItem>
   405→                    <SelectItem value="mmCIF">mmCIF</SelectItem>
   406→                  </SelectContent>
   407→                </Select>
   408→              </div>
   409→              <div>
   410→                <label className="text-sm font-medium">Source URL</label>
   411→                <Input
   412→                  value={formData.sourceUrl}
   413→                  onChange={(e) => setFormData({...formData, sourceUrl: e.target.value})}
   414→                  placeholder="https://..."
   415→                  className="mt-1"
   416→                />
   417→              </div>
   418→            </div>
   419→
   420→            {/* Upload Progress */}
   421→            {isUploading && (
   422→              <div className="mb-4">
   423→                <div className="flex justify-between text-sm mb-2">
   424→                  <span>Uploading...</span>
   425→                  <span>{uploadProgress}%</span>
   426→                </div>
   427→                <Progress value={uploadProgress} />
   428→              </div>
   429→            )}
   430→
   431→            <div className="flex justify-end gap-3">
   432→              <Button variant="outline" onClick={() => setShowCreateForm(false)} disabled={isUploading}>
   433→                Cancel
   434→              </Button>
   435→              <Button onClick={handleCreateDataset} disabled={!formData.name.trim() || isUploading}>
   436→                {isUploading ? '⏳ Uploading...' : '📤 Upload Dataset'}
   437→              </Button>
   438→            </div>
   439→          </CardContent>
   440→        </Card>
   441→      )}
   442→
   443→      {/* Datasets Display */}
   444→      {viewMode === 'grid' ? (
   445→        /* Grid View */
   446→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
   447→          {filteredDatasets.map((dataset) => (
   448→            <Card 
   449→              key={dataset.id} 
   450→              className={`hover:shadow-lg transition-all hover:-translate-y-1 cursor-pointer ${
   451→                selectedDataset === dataset.id ? 'ring-2 ring-primary' : ''
   452→              }`}
   453→              onClick={() => setSelectedDataset(dataset.id)}
   454→            >
   455→              <CardHeader className="pb-3">
   456→                <div className="flex items-start justify-between">
   457→                  <div className="flex-1 min-w-0">
   458→                    {editingField?.id === dataset.id && editingField.field === 'name' ? (
   459→                      <Input
   460→                        value={tempValue}
   461→                        onChange={(e) => setTempValue(e.target.value)}
   462→                        onBlur={() => saveEdit(dataset.id, 'name')}
   463→                        onKeyDown={(e) => e.key === 'Enter' && saveEdit(dataset.id, 'name')}
   464→                        className="h-7 text-lg font-medium"
   465→                        autoFocus
   466→                      />
   467→                    ) : (
   468→                      <CardTitle 
   469→                        className="text-lg truncate"
   470→                        onDoubleClick={(e) => {
   471→                          e.stopPropagation();
   472→                          startEditing(dataset.id, 'name', dataset.name.value);
   473→                        }}
   474→                        title="Double-click to edit"
   475→                      >
   476→                        {dataset.name.value}
   477→                        {dataset.name.isDirty && <span className="ml-1 text-orange-500">✏️</span>}
   478→                      </CardTitle>
   479→                    )}
   480→                    <CardDescription className="text-xs mt-1 line-clamp-2">
   481→                      {dataset.description.value}
   482→                    </CardDescription>
   483→                  </div>
   484→                  
   485→                  <div className="flex flex-col items-end gap-1">
   486→                    <Badge variant="outline" className="text-xs">
   487→                      {dataset.type.value}
   488→                    </Badge>
   489→                    <Badge variant={dataset.isPublic.value ? 'default' : 'secondary'} className="text-xs">
   490→                      {dataset.isPublic.value ? 'Public' : 'Private'}
   491→                    </Badge>
   492→                  </div>
   493→                </div>
   494→              </CardHeader>
   495→              
   496→              <CardContent className="space-y-3">
   497→                {/* Info Grid */}
   498→                <div className="grid grid-cols-2 gap-2 text-xs">
   499→                  <div className="flex items-center gap-1">
   500→                    <span>📦</span>
   501→                    <span>{dataset.size.value}</span>
   502→                  </div>
   503→                  <div className="flex items-center gap-1">
   504→                    <span>📊</span>
   505→                    <span>{dataset.rows.value.toLocaleString()} rows</span>
   506→                  </div>
   507→                  <div className="flex items-center gap-1">
   508→                    <span>📋</span>
   509→                    <span>{dataset.columns.value} cols</span>
   510→                  </div>
   511→                  <div className="flex items-center gap-1">
   512→                    <span>🏷️</span>
   513→                    <span>{dataset.format.value}</span>
   514→                  </div>
   515→                </div>
   516→
   517→                {/* Tags */}
   518→                <div className="flex flex-wrap gap-1">
   519→                  {dataset.tags.value.slice(0, 3).map((tag, idx) => (
   520→                    <Badge key={idx} variant="secondary" className="text-xs">
   521→                      {tag}
   522→                    </Badge>
   523→                  ))}
   524→                  {dataset.tags.value.length > 3 && (
   525→                    <Badge variant="secondary" className="text-xs">
   526→                      +{dataset.tags.value.length - 3}
   527→                    </Badge>
   528→                  )}
   529→                </div>
   530→
   531→                {/* Storage Location */}
   532→                <div className="pt-2 border-t">
   533→                  <div className="flex items-center justify-between text-xs">
   534→                    <span className="text-muted-foreground">Storage:</span>
   535→                    <Badge variant="outline" className="text-xs">
   536→                      {dataset.storageLocation === 'duckdb' && '🦆 DuckDB'}
   537→                      {dataset.storageLocation === 'postgres' && '🐘 PostgreSQL'}
   538→                      {dataset.storageLocation === 'indexeddb' && '🗄️ IndexedDB'}
   539→                      {dataset.storageLocation === 'local' && '💾 Local'}
   540→                    </Badge>
   541→                  </div>
   542→                </div>
   543→
   544→                {/* Actions */}
   545→                <div className="flex gap-2 pt-2 border-t">
   546→                  <Button 
   547→                    size="sm" 
   548→                    variant="outline" 
   549→                    className="flex-1"
   550→                    onClick={(e) => {
   551→                      e.stopPropagation();
   552→                      handleQueryDataset(dataset.id);
   553→                    }}
   554→                  >
   555→                    🔍 Query
   556→                  </Button>
   557→                  <Button 
   558→                    size="sm" 
   559→                    variant="outline"
   560→                    onClick={(e) => {
   561→                      e.stopPropagation();
   562→                      handleExportDataset(dataset.id, 'csv');
