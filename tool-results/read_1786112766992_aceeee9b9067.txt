   107→// Confidence Score Badge
   108→function ConfidenceBadge({ score, size = 'default' }: { score: number; size?: 'sm' | 'default' | 'lg' }) {
   109→  const getColor = () => {
   110→    if (score >= 90) return 'bg-green-100 text-green-800 border-green-300';
   111→    if (score >= 70) return 'bg-yellow-100 text-yellow-800 border-yellow-300';
   112→    return 'bg-red-100 text-red-800 border-red-300';
   113→  };
   114→
   115→  const getLabel = () => {
   116→    if (score >= 90) return 'High Confidence';
   117→    if (score >= 70) return 'Medium Confidence';
   118→    return 'Low Confidence';
   119→  };
   120→
   121→  const sizeClasses = {
   122→    sm: 'px-2 py-0.5 text-xs',
   123→    default: 'px-3 py-1 text-sm',
   124→    lg: 'px-4 py-2 text-base'
   125→  };
   126→
   127→  return (
   128→    <Badge variant="outline" className={`${getColor()} ${sizeClasses[size]}`}>
   129→      {score.toFixed(1)}% • {getLabel()}
   130→    </Badge>
   131→  );
   132→}

   152→// Simplified 3D Molecule Visualization
   153→function ProteinStructureViewer({ protein, confidenceScores }: ProteinVisualizationProps) {
   154→  const [rotation, setRotation] = useState(0);
   155→  const [showConfidence, setShowConfidence] = useState(true);
   156→
   157→  // Auto-rotate animation
   158→  useEffect(() => {
   159→    const interval = setInterval(() => {
   160→      setRotation(r => (r + 0.5) % 360);
   161→    }, 50);
   162→    return () => clearInterval(interval);
   163→  }, []);
   164→
   165→  // Generate simplified atom positions based on sequence length
   166→  const generateAtoms = () => {
   167→    const count = Math.min(protein.sequenceLength || 200, 150);
   168→    const atoms = [];
   169→    
   170→    for (let i = 0; i < count; i++) {
   171→      const angle = (i / count) * Math.PI * 4 + (rotation * Math.PI / 180);
   172→      const radius = 30 + Math.sin(i * 0.1) * 20;
   173→      const height = (i / count) * 120 - 60;
   174→      
   175→      const x = Math.cos(angle) * radius;
   176→      const y = height;
   177→      const z = Math.sin(angle) * radius;
   178→      
   179→      const confidence = confidenceScores?.[i % (confidenceScores?.length || 1)] || (protein.confidenceScore || 80);
   180→      
   181→      atoms.push({
   182→        id: i,
   183→        x, y, z,
   184→        confidence,
   185→        type: i % 3 === 0 ? 'alpha' : i % 3 === 1 ? 'beta' : 'coil'
   186→      });
   187→    }
   188→    
   189→    return atoms;
   190→  };
   191→
   192→  const atoms = generateAtoms();
   193→  
   194→  const getAtomColor = (atom: typeof atoms[0]) => {
   195→    if (!showConfidence) {
   196→      return atom.type === 'alpha' ? '#3b82f6' : atom.type === 'beta' ? '#ef4444' : '#22c55e';
   197→    }
   198→    if (atom.confidence >= 90) return '#22c55e'; // Green for high
   199→    if (atom.confidence >= 70) return '#eab308'; // Yellow for medium
   200→    return '#ef4444'; // Red for low
   201→  };
   202→
   203→  return (
   204→    <div className="relative w-full h-[400px] bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 rounded-lg overflow-hidden">
   205→      {/* Grid background */}
   206→      <div className="absolute inset-0 opacity-10" style={{
   207→        backgroundImage: `
   208→          linear-gradient(rgba(255,255,255,0.1) 1px, transparent 1px),
   209→          linear-gradient(90deg, rgba(255,255,255,0.1) 1px, transparent 1px)
   210→        `,
   211→        backgroundSize: '40px 40px'
   212→      }} />
   213→      
   214→      {/* SVG Visualization */}
   215→      <svg viewBox="-100 -80 200 160" className="w-full h-full" style={{ transform: 'rotateX(15deg)' }}>
   216→        {/* Bonds between nearby atoms */}
   217→        {atoms.map((atom, i) => 
   218→          i > 0 && (
   219→            <line
   220→              key={`bond-${i}`}
   221→              x1={atoms[i-1].x}
   222→              y1={atoms[i-1].y}
   223→              x2={atom.x}
   224→              y2={atom.y}
   225→              stroke="rgba(255,255,255,0.2)"
   226→              strokeWidth="0.5"
   227→            />
   228→          )
   229→        )}
   230→        
   231→        {/* Atoms */}
   232→        {atoms.map((atom) => (
   233→          <circle
   234→            key={atom.id}
   235→            cx={atom.x}
   236→            cy={atom.y}
   237→            r={atom.type === 'alpha' ? 2.5 : atom.type === 'beta' ? 2 : 1.5}
   238→            fill={getAtomColor(atom)}
   239→            opacity={0.9}
   240→          />
   241→        ))}
   242→        
   243→        {/* Center label */}
   244→        <text x="0" y="75" textAnchor="middle" fill="rgba(255,255,255,0.6)" fontSize="6" fontFamily="monospace">
   245→          {protein.uniprotId}
   246→        </text>
   247→      </svg>
   248→
   249→      {/* Controls overlay */}
   250→      <div className="absolute bottom-4 left-4 right-4 flex justify-between items-center">
   251→        <div className="flex gap-2">
   252→          <Button
   253→            variant="secondary"
   254→            size="sm"
   255→            onClick={() => setShowConfidence(!showConfidence)}
   256→            className="bg-black/50 hover:bg-black/70 text-white border-white/20"
   257→          >
   258→            {showConfidence ? 'Show Type' : 'Show Confidence'}
   259→          </Button>
   260→        </div>
   261→        
   262→        <Badge variant="outline" className="bg-black/50 text-white border-white/20">
   263→          {atoms.length} residues visualized
   264→        </Badge>
   265→      </div>
   266→
   267→      {/* Legend */}
   268→      <div className="absolute top-4 right-4 bg-black/50 rounded-lg p-3 backdrop-blur-sm">
   269→        <p className="text-xs text-white font-semibold mb-2">Legend</p>
   270→        {showConfidence ? (
   271→          <>
   272→            <div className="flex items-center gap-2 mb-1">
   273→              <div className="w-3 h-3 rounded-full bg-green-500" />
   274→              <span className="text-xs text-gray-300">High (&ge;90%)</span>
   275→            </div>
   276→            <div className="flex items-center gap-2 mb-1">
   277→              <div className="w-3 h-3 rounded-full bg-yellow-500" />
   278→              <span className="text-xs text-gray-300">Medium (70-90%)</span>
   279→            </div>
   280→            <div className="flex items-center gap-2">
   281→              <div className="w-3 h-3 rounded-full bg-red-500" />
   282→              <span className="text-xs text-gray-300">Low (&lt;70%)</span>
   283→            </div>
   284→          </>
   285→        ) : (
   286→          <>
   287→            <div className="flex items-center gap-2 mb-1">
   288→              <div className="w-3 h-3 rounded-full bg-blue-500" />
   289→              <span className="text-xs text-gray-300">Alpha helix</span>
   290→            </div>
   291→            <div className="flex items-center gap-2 mb-1">
   292→              <div className="w-3 h-3 rounded-full bg-red-500" />
   293→              <span className="text-xs text-gray-300">Beta sheet</span>
   294→            </div>
   295→            <div className="flex items-center gap-2">
   296→              <div className="w-3 h-3 rounded-full bg-green-500" />
   297→              <span className="text-xs text-gray-300">Coil</span>
   298→            </div>
   299→          </>
   300→        )}
   301→      </div>
   302→    </div>
   303→  );
   304→}

   306→// Confidence Score Chart
   307→function ConfidenceChart({ scores }: { scores: number[] }) {
   308→  const maxScore = 100;
   309→  const chartHeight = 80;
   310→  const barWidth = 3;
   311→  const gap = 1;
   312→
   313→  return (
   314→    <div className="w-full overflow-x-auto py-2">
   315→      <div className="inline-flex items-end gap-[1px]" style={{ minHeight: chartHeight }}>
   316→        {scores.slice(0, 100).map((score, i) => {
   317→          const height = (score / maxScore) * chartHeight;
   318→          const color = score >= 90 ? '#22c55e' : score >= 70 ? '#eab308' : '#ef4444';
   319→          
   320→          return (
   321→            <TooltipProvider key={i}>
   322→              <Tooltip>
   323→                <TooltipTrigger asChild>
   324→                  <div
   325→                    className="transition-all hover:opacity-80"
   326→                    style={{
   327→                      width: barWidth,
   328→                      height: `${height}px`,
   329→                      backgroundColor: color,
   330→                      borderRadius: '1px'
   331→                    }}
   332→                  />
   333→                </TooltipTrigger>
   334→                <TooltipContent side="top">
   335→                  <p className="text-xs">Residue {i + 1}: {score.toFixed(1)}%</p>
   336→                </TooltipContent>
   337→              </Tooltip>
   338→            </TooltipProvider>
   339→          );
   340→        })}
   341→      </div>
   342→      <div className="flex justify-between mt-1 text-xs text-muted-foreground">
   343→        <span>1</span>
   344→        <span>Position</span>
   345→        <span>{Math.min(scores.length, 100)}</span>
   346→      </div>
   347→    </div>
   348→  );
   349→}

   420→// Main AlphaFold Component
   421→export function AlphaFoldConnector() {
   422→  // State management
   423→  const [activeTab, setActiveTab] = useState('search');
   424→  const [searchQuery, setSearchQuery] = useState('');
   425→  const [isSearching, setIsSearching] = useState(false);
   426→  const [searchResults, setSearchResults] = useState<AlphaFoldSearchResult | null>(null);
   427→  const [selectedProtein, setSelectedProtein] = useState<AlphaFoldProtein | null>(null);
   428→  const [structureData, setStructureData] = useState<any>(null);
   429→  const [isLoadingStructure, setIsLoadingStructure] = useState(false);
   430→  const [featuredProteins, setFeaturedProteins] = useState<AlphaFoldProtein[]>([]);
   431→  const [dbStats, setDbStats] = useState<any>(null);
   432→  const [batchInput, setBatchInput] = useState('');
   433→  const [batchResults, setBatchResults] = useState<BatchQueryResult | null>(null);
   434→  const [isBatchProcessing, setIsBatchProcessing] = useState(false);
   435→
   436→  // Load featured proteins and stats on mount
   437→  useEffect(() => {
   438→    loadInitialData();
   439→  }, []);
   440→
   441→  const loadInitialData = async () => {
   442→    try {
   443→      const [proteins, stats] = await Promise.all([
   444→        getFeaturedProteins(),
   445→        getDatabaseStats()
   446→      ]);
   447→      setFeaturedProteins(proteins);
   448→      setDbStats(stats);
   449→    } catch (error) {
   450→      console.error('Error loading initial data:', error);
   451→    }
   452→  };
   453→
   454→  // Search handler
   455→  const handleSearch = async () => {
   456→    if (!searchQuery.trim()) return;
   457→    
   458→    setIsSearching(true);
   459→    try {
   460→      // Determine if it's a UniProt ID (starts with letter followed by numbers/letters)
   461→      const isUniProtId = /^[A-Z][0-9A-Z]{1,10}$/i.test(searchQuery.trim());
   462→      
   463→      let results: AlphaFoldSearchResult;
   464→      
   465→      if (isUniProtId) {
   466→        // Direct UniProt ID lookup
   467→        const protein = await fetchProteinByUniProt(searchQuery.trim());
   468→        results = {
   469→          query: searchQuery.trim(),
   470→          results: [protein],
   471→          totalFound: 1,
   472→          searchTime: Date.now(),
   473→          source: 'alphafold'
   474→        };
   475→      } else {
   476→        // Keyword search
   477→        results = await searchProteins(searchQuery.trim());
   478→      }
   479→      
   480→      setSearchResults(results);
   481→    } catch (error) {
   482→      console.error('Search error:', error);
   483→    } finally {
   484→      setIsSearching(false);
   485→    }
   486→  };
   487→
   488→  // View protein details
   489→  const handleViewDetails = async (protein: AlphaFoldProtein) => {
   490→    setSelectedProtein(protein);
   491→    setIsLoadingStructure(true);
   492→    
   493→    try {
   494→      const structure = await fetchStructureData(protein.uniprotId);
   495→      setStructureData(structure);
   496→    } catch (error) {
   497→      console.error('Error loading structure:', error);
   498→    } finally {
   499→      setIsLoadingStructure(false);
   500→    }
   501→  };
   502→
   503→  // Batch query handler
   504→  const handleBatchQuery = async () => {
   505→    const ids = batchInput
   506→      .split(/[\n,]+/)
   507→      .map(id => id.trim())
   508→      .filter(id => id.length > 0);
   509→    
   510→    if (ids.length === 0) return;
   511→    
   512→    setIsBatchProcessing(true);
   513→    try {
   514→      const results = await batchQueryProteins(ids);
   515→      setBatchResults(results);
   516→    } catch (error) {
   517→      console.error('Batch query error:', error);
   518→    } finally {
   519→      setIsBatchProcessing(false);
   520→    }
   521→  };
   522→
   523→  // Download PDB handler
   524→  const handleDownloadPDB = (protein: AlphaFoldProtein) => {
   525→    const pdbContent = formatPDBDownload(protein, structureData?.pdbContent);
   526→    const blob = new Blob([pdbContent], { type: 'text/plain' });
   527→    const url = URL.createObjectURL(blob);
   528→    const a = document.createElement('a');
   529→    a.href = url;
   530→    a.download = `${protein.entryId}-alphafold.pdb`;
   531→    document.body.appendChild(a);
   532→    a.click();
   533→    document.body.removeChild(a);
   534→    URL.revokeObjectURL(url);
   535→  };
   536→
   537→  // Copy citation handler
   538→  const handleCopyCitation = (protein: AlphaFoldProtein) => {
   539→    const citation = generateCitation(protein);
   540→    navigator.clipboard.writeText(citation);
   541→  };
   542→
   543→  return (
   544→    <div className="space-y-6">
   545→      {/* Header Card */}
   546→      <Card className="border-2 border-blue-200 bg-gradient-to-br from-blue-50 to-purple-50 dark:from-blue-950/20 dark:to-purple-950/20">
   547→        <CardHeader>
   548→          <div className="flex items-start justify-between">
   549→            <div className="space-y-2">
   550→              <div className="flex items-center gap-3">
   551→                <div className="p-3 bg-gradient-to-br from-blue-500 to-purple-600 rounded-xl">
   552→                  <Dna className="h-8 w-8 text-white" />
   553→                </div>
   554→                <div>
   555→                  <CardTitle className="text-2xl flex items-center gap-2">
   556→                    AlphaFold DB Connector
   557→                    <Badge className="bg-green-100 text-green-800 border-green-300">
   558→                      <Zap className="h-3 w-3 mr-1" />
   559→                      FREE TIER
   560→                    </Badge>
   561→                  </CardTitle>
   562→                  <CardDescription className="text-base mt-1">
   563→                    Google DeepMind&apos;s Revolutionary AI Protein Structure Prediction
   564→                  </CardDescription>
   565→                </div>
   566→              </div>
   567→              
   568→              <p className="text-sm text-muted-foreground max-w-3xl">
   569→                Access 200M+ predicted protein structures with Nobel Prize-level accuracy. 
   570→                No API key required. Free for research and educational use.
   571→                Powered by artificial intelligence that solved the 50-year grand challenge of protein folding.
   572→              </p>
   573→            </div>
   574→            
   575→            <div className="hidden md:flex flex-col gap-2 min-w-[160px]">
   576→              <a 
   577→                href="https://alphafold.ebi.ac.uk/" 
   578→                target="_blank" 
   579→                rel="noopener noreferrer"
   580→                className="text-sm text-blue-600 hover:text-blue-800 flex items-center gap-1"
   581→              >
   582→                <Globe className="h-4 w-4" />
   583→                alphafold.ebi.ac.uk
   584→              </a>
   585→              <a 
   586→                href="https://www.nature.com/articles/s41586-021-03819-2"
