    95→    95→  const [saveNotes, setSaveNotes] = useState('');

   104→   104→    'CRISPR gene editing cancer therapy'

   106→   106→

   117→   117→    const queryToUse = searchQuery || syntheticQuery;

   137→   137→        filters: filters as Record<string, unknown>,
   138→   138→      });
   139→   139→
   140→   140→      // Update dashboard stats

   142→   142→
   143→   143→      // Log activity

   145→   145→        type: 'search',
   146→   146→        message: createDynamicField(
   147→   147→          `Searched "${queryToUse.substring(0, 50)}${queryToUse.length > 50 ? '...' : ''}" — ${response.total} results (${response.source})`
   148→   148→        ),
   149→   149→        icon: response.source === 'real-apis' ? '🔍' : '🤖',
   150→   150→        metadata: { source: response.source, queryTime: response.queryTime },
   151→   151→      });
   152→   152→
   153→   153→      // Show call-for-action if user is getting good results
   154→   154→      if (response.total > 50 && savedItems.length < 5) {
   155→   155→        setTimeout(() => {
   156→   156→          store.triggerUpgradePrompt('api_rate');
   157→   157→        }, 2000);
   158→   158→      }
   159→   159→
   160→   160→    } catch (error) {
   161→   161→      console.error('Search failed:', error);
   162→   162→      
   163→   163→      // NEVER let user see an error without help
   164→   164→      addActivity({
   165→   165→        type: 'error_recovery',
   166→   166→        message: createDynamicField('Search encountered an issue — showing cached/synthetic results'),
   167→   167→        icon: '⚠️',
   168→   168→      });
   169→   169→      
   170→   170→      // Show synthetic results anyway (graceful degradation)
   171→   171→      setSearchResults([]);
   172→   172→      setTotalResults(0);
   173→   173→    } finally {
   174→   174→      setIsSearching(false);
   175→   175→    }
   176→   176→  }, [searchQuery, syntheticQuery, filters]);
   177→   177→
   178→   178→  // Auto-search on Enter or button click
   179→   179→  useEffect(() => {
   180→   180→    const handler = (e: KeyboardEvent) => {
   181→   181→      if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
   182→   182→        handleSearch();
   183→   183→      }
   184→   184→    };
   185→   185→    window.addEventListener('keydown', handler);
   186→   186→    return () => window.removeEventListener('keydown', handler);
   187→   187→  }, [handleSearch]);
   188→   188→
   189→   189→  // ============ SAVE HANDLER ============
   190→   190→
   191→   191→  const handleSaveResult = (result: SearchResult) => {
   192→   192→    setSelectedResult(result);
   193→   193→    setShowSaveDialog(true);
   194→   194→  };
   195→   195→
   196→   196→  const confirmSave = () => {
   197→   197→    if (!selectedResult) return;
   198→   198→
   199→   199→    saveItem({
   200→   200→      type: 'paper',
   201→   201→      title: selectedResult.title,
   202→   202→      source: selectedResult.source,
   203→   203→      metadata: {
   204→   204→        authors: selectedResult.authors,
   205→   205→        year: selectedResult.year,
   206→   206→        doi: selectedResult.doi,
   207→   207→        url: selectedResult.url,
   208→   208→        abstract: selectedResult.abstract,
   209→   209→      },
   210→   210→      tags: saveTags.split(',').map(t => t.trim()).filter(Boolean),
   211→   211→      notes: saveNotes || undefined,
   212→   212→    });
   213→   213→
   214→   214→    addActivity({
   215→   215→      type: 'save',
   216→   216→      message: createDynamicField(`Saved: ${selectedResult.title.substring(0, 50)}...`),
   217→   217→      icon: '⭐',
   218→   218→    });
   219→   219→
   220→   220→    setShowSaveDialog(false);
   221→   221→    setSelectedResult(null);
   222→   222→    setSaveTags('');
   223→   223→    setSaveNotes('');
   224→   224→  };
   225→   225→
   226→   226→  // ============ SAVE QUERY HANDLER ============
   227→   227→
   228→   228→  const handleSaveQuery = () => {
   229→   229→    const queryText = searchQuery || syntheticQuery;
   230→   230→    if (!queryText.trim()) return;
   231→   231→
   232→   232→    addSavedQuery({
   233→   233→      name: createDynamicField(queryText.substring(0, 40)),
   234→   234→      sql: createDynamicField(`SELECT * FROM papers WHERE MATCH('${queryText}')`), // Simplified SQL representation
   235→   235→      description: createDynamicField(`Search: ${queryText}`),
   236→   236→      runCount: createDynamicField(1),
   237→   237→      dataSource: searchSource,
   238→   238→    });
   239→   239→
   240→   240→    addActivity({
   241→   241→      type: 'save',
   242→   242→      message: createDynamicField(`Saved query: ${queryText.substring(0, 40)}...`),
   243→   243→      icon: '💾',
   244→   244→    });
   245→   245→  };
   246→   246→
   247→   247→  // ============ EXPORT HANDLER ============
   248→   248→
   249→   249→  const handleExportResults = (format: 'csv' | 'json' | 'bib') => {
   250→   250→    if (searchResults.length === 0) return;
   251→   251→
   252→   252→    let content: string;
   253→   253→    let filename: string;
   254→   254→    let mimeType: string;
   255→   255→
   256→   256→    switch (format) {
   257→   257→      case 'csv':
   258→   258→        const headers = ['Title', 'Authors', 'Year', 'DOI', 'Journal', 'Citations'];
   259→   259→        const rows = searchResults.map(r => [
   260→   260→          `"${r.title.replace(/"/g, '""')}"`,
   261→   261→          `"${r.authors.join('; ')}"`,
   262→   262→          r.year,
   263→   263→          r.doi || '',
   264→   264→          r.journal || '',
   265→   265→          r.citations || '',

   267→   267→        content = [headers.join(','), ...rows.map(r => r.join(','))].join('\n');
   268→   268→        filename = `scihub-results-${Date.now()}.csv`;
   269→   269→        mimeType = 'text/csv';
   270→   270→        break;
   271→   271→
   272→   272→      case 'json':
   273→   273→        content = JSON.stringify(searchResults, null, 2);
   274→   274→        filename = `scihub-results-${Date.now()}.json`;
   275→   275→        mimeType = 'application/json';
   276→   276→        break;
   277→   277→
   278→   278→      case 'bib':
   279→   279→        content = searchResults.map(r => 
   280→   280→          `@article{${r.doi?.replace(/\//g, '') || r.id},\n` +
   281→   281→          `  title = {${r.title}},\n` +
   282→   282→          `  author = {${r.authors.join(' and ')}},\n` +
   283→   283→          `  year = {${r.year}},\n` +
   284→   284→          `  journal = {${r.journal || 'Unknown'}},\n` +
   285→   285→          `  doi = {${r.doi || ''}},\n` +
   286→   286→          `}\n`
   287→   287→        ).join('\n');
   288→   288→        filename = `scihub-results-${Date.now()}.bib`;
   289→   289→        mimeType = 'text/x-bibtex';
   290→   290→        break;
   291→   291→    }
   292→   292→
   293→   293→    // Download file
   294→   294→    const blob = new Blob([content], { type: mimeType });
   295→   295→    const url = URL.createObjectURL(blob);
   296→   296→    const a = document.createElement('a');
   297→   297→    a.href = url;
   298→   298→    a.download = filename;
   299→   299→    a.click();
   300→   300→    URL.revokeObjectURL(url);
   301→   301→
   302→   302→    addActivity({
   303→   303→      type: 'export',
   304→   304→      message: createDynamicField(`Exported ${searchResults.length} results as ${format.toUpperCase()}`),
   305→   305→      icon: '📥',
   306→   306→    });
   307→   307→
   308→   308→    // Call-for-action for premium formats
   309→   309→    if (format === 'bib') {
   310→   310→      store.triggerUpgradePrompt('export_format');
   311→   311→    }
   312→   312→  };
   313→   313→
   314→   314→  // ============ RENDER HELPERS ============
   315→   315→
   316→   316→  const formatAuthors = (authors: string[]) => {
   317→   317→    if (authors.length <= 3) return authors.join(', ');
   318→   318→    return `${authors.slice(0, 3).join(', ')} et al.`;
   319→   319→  };
   320→   320→
   321→   321→  const getSourceBadge = (source: SearchResult['source']) => {

   323→   323→      crossref: 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-200',
   324→   324→      openalex: 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200',
   325→   325→      arxiv: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
   326→   326→      ncbi: 'bg-purple-100 text-purple-800 dark:bg-purple-900 dark:text-purple-200',
   327→   327→      synthetic: 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-200',
   328→   328→    };
   329→   329→    return styles[source] || styles.synthetic;
   330→   330→  };
   331→   331→
   332→   332→  return (
   333→   333→    <div className="min-h-screen bg-background p-6">
   334→   334→      {/* Header */}
   335→   335→      <div className="mb-8">
   336→   336→        <h1 className="text-3xl font-bold text-foreground">
   337→   337→          {t('query.title') || 'Scientific Literature Search'}
   338→   338→        </h1>
   339→   339→        <p className="text-muted-foreground mt-1">

   341→   341→        </p>
   342→   342→        
   343→   343→        {/* Free Tier Indicator */}
   344→   344→        <div className="mt-3 inline-flex items-center gap-2 px-3 py-1 bg-green-50 dark:bg-green-950 border border-green-200 dark:border-green-800 rounded-full text-sm">
   345→   345→          <span className="text-green-600 dark:text-green-400">🆓</span>
   346→   346→          <span className="text-green-800 dark:text-green-200">
   347→   347→            Free tier active • Unlimited searches • Real API access
   348→   348→          </span>
   349→   349→        </div>
   350→   350→      </div>
   351→   351→
   352→   352→      {/* Guidance Panel (Progressive Discovery) */}
   353→   353→      {showGuidancePanel && relevantGuidance.length > 0 && (
   354→   354→        <Card className="mb-6 border-blue-200 dark:border-blue-800 bg-blue-50/50 dark:bg-blue-950/30">
   355→   355→          <CardContent className="p-4">
   356→   356→            <div className="flex items-start justify-between">
   357→   357→              <div className="flex items-start gap-3">
   358→   358→                <span className="text-2xl">{relevantGuidance[0].icon}</span>
   359→   359→                <div>
   360→   360→                  <h4 className="font-medium text-blue-900 dark:text-blue-100">
   361→   361→                    {relevantGuidance[0].title}
   362→   362→                  </h4>
   363→   363→                  <p className="text-sm text-blue-700 dark:text-blue-300 mt-1">
   364→   364→                    {relevantGuidance[0].message}
   365→   365→                  </p>
   366→   366→                  {relevantGuidance[0].targetRoute && (
   367→   367→                    <Button
   368→   368→                      size="sm"
   369→   369→                      variant="link"
   370→   370→                      className="mt-2 p-0 h-auto text-blue-600 dark:text-blue-400"
   371→   371→                      onClick={() => {
   372→   372→                        window.location.href = relevantGuidance[0].targetRoute;
   373→   373→                        dismissGuidance(relevantGuidance[0].id);
   374→   374→                      }}
   375→   375→                    >
   376→   376→                      Try it now →
   377→   377→                    </Button>
   378→   378→                  )}
   379→   379→                </div>
   380→   380→              </div>
   381→   381→              <Button
   382→   382→                size="sm"
   383→   383→                variant="ghost"
   384→   384→                onClick={() => {
   385→   385→                  dismissGuidance(relevantGuidance[0].id);
   386→   386→                  if (relevantGuidance.length <= 1) setShowGuidancePanel(false);
   387→   387→                }}
   388→   388→              >
   389→   389→                ✕
   390→   390→              </Button>
   391→   391→            </div>
   392→   392→          </CardContent>
   393→   393→        </Card>
   394→   394→      )}
   395→   395→
   396→   396→      {/* Main Content */}
   397→   397→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   398→   398→        <TabsList className="mb-6">
   399→   399→          <TabsTrigger value="search">🔍 Search</TabsTrigger>
   400→   400→          <TabsTrigger value="saved">⭐ Saved ({savedItems.length})</TabsTrigger>

   402→   402→          <TabsTrigger value="history">📜 History</TabsTrigger>
   403→   403→        </TabsList>
   404→   404→
   405→   405→        {/* SEARCH TAB */}
   406→   406→        <TabsContent value="search" className="space-y-6">
   407→   407→          {/* Search Input Area */}
   408→   408→          <Card>
   409→   409→            <CardContent className="p-6">
   410→   410→              <div className="space-y-4">
   411→   411→                {/* Main Search Input */}
   412→   412→                <div className="flex gap-4">
   413→   413→                  <div className="flex-1 relative">
   414→   414→                    <Input
   415→   415→                      placeholder={
   416→   416→                        searchQuery ? '' : 
   417→   417→                        "Try: CRISPR gene editing cancer therapy (or type your own query)"
   418→   418→                      }
   419→   419→                      value={searchQuery || (!searchQuery ? syntheticQuery : '')}
   420→   420→                      onChange={(e) => {
   421→   421→                        setSearchQuery(e.target.value);
   422→   422→                        if (e.target.value) setSyntheticQuery(''); // Clear synthetic on first keystroke
   423→   423→                      }}
   424→   424→                      className={`text-lg h-12 px-4 ${
   425→   425→                        !searchQuery && syntheticQuery ? 'text-muted-foreground italic' : ''
   426→   426→                      }`}
   427→   427→                      onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   428→   428→                    />
   429→   429→                    {!searchQuery && syntheticQuery && (
   430→   430→                      <span className="absolute right-3 top-3 text-xs text-muted-foreground bg-muted px-2 py-1 rounded">
   431→   431→                        Suggested query • Edit to replace
   432→   432→                      </span>
   433→   433→                    )}
   434→   434→                  </div>
   435→   435→                  
   436→   436→                  <Select value={searchSource} onValueChange={setSearchSource}>
   437→   437→                    <SelectTrigger className="w-[180px]">
   438→   438→                      <SelectValue />
   439→   439→                    </SelectTrigger>
   440→   440→                    <SelectContent>

   442→   442→                      <SelectItem value="crossref">CrossRef</SelectItem>
   443→   443→                      <SelectItem value="openalex">OpenAlex</SelectItem>
   444→   444→                      <SelectItem value="arxiv">arXiv</SelectItem>
   445→   445→                      <SelectItem value="ncbi">NCBI PubMed</SelectItem>
   446→   446→                    </SelectContent>
   447→   447→                  </Select>
   448→   448→
   449→   449→                  <Button
   450→   450→                    onClick={handleSearch}
   451→   451→                    disabled={isSearching || (!searchQuery && !syntheticQuery)}
   452→   452→                    size="lg"
   453→   453→                    className="px-8"
   454→   454→                  >

   456→   456→                      <>

   460→   460→                    ) : (
   461→   461→                      <>🔍 Search</>
   462→   462→                    )}
   463→   463→                  </Button>
   464→   464→                </div>
   465→   465→
   466→   466→                {/* Filters Row */}

   478→   478→                    <Input
   479→   479→                      type="number"
   480→   480→                      placeholder="To"
