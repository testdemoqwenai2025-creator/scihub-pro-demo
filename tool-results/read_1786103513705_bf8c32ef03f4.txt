   641→  const container = document.getElementById(containerId);
   642→  
   643→  // Setup SVG
   644→  const width = 800;
   645→  const height = 500;
   646→  const margin = { top: 40, right: 40, bottom: 60, left: 70 };
   647→  
   648→  // Scales
   649→  const xScale = d3.scaleLinear()
   650→    .domain(d3.extent(data.genes, d => d.foldChange))
   651→    .range([margin.left, width - margin.right]);
   652→    
   653→  const yScale = d3.scaleLinear()
   654→    .domain([0, -Math.log10(d3.min(data.genes, d => d.pValue)) * 1.1])
   655→    .range([height - margin.bottom, margin.top]);
   656→  
   657→  // Draw points
   658→  const svg = d3.select(container).append('svg')
   659→    .attr('width', width)
   660→    .attr('height', height);
   661→    
   662→  svg.selectAll('circle')
   663→    .data(data.genes)
   664→    .enter()
   665→    .append('circle')
   666→    .attr('cx', d => xScale(d.foldChange))
   667→    .attr('cy', d => yScale(-Math.log10(d.pValue)))
   668→    .attr('r', d => d.significant ? 6 : 4)
   669→    .attr('fill', d => {
   670→      if (d.significant && d.foldChange > 0) return '#e41a1c'; // Red up
   671→      if (d.significant && d.foldChange < 0) return '#377eb8'; // Blue down
   672→      return '#999999'; // Gray not significant
   673→    })
   674→    .attr('opacity', 0.7)
   675→    .on('mouseover', showTooltip)
   676→    .on('mouseout', hideTooltip);
   677→    
   678→  // Threshold lines
   679→  svg.append('line')
   680→    .attr('x1', xScale(-1))
   681→    .attr('y1', margin.top)
   682→    .attr('x2', xScale(-1))
   683→    .attr('y2', height - margin.bottom)
   684→    .attr('stroke', 'green')
   685→    .attr('stroke-dasharray', '5,5');
   686→    
   687→  svg.append('line')
   688→    .attr('x1', xScale(1))
   689→    .attr('y1', margin.top)
   690→    .attr('x2', xScale(1))
   691→    .attr('y2', height - margin.bottom)
   692→    .attr('stroke', 'green')
   693→    .attr('stroke-dasharray', '5,5');
   694→    
   695→  svg.append('line')
   696→    .attr('x1', margin.left)
   697→    .attr('y1', yScale(-Math.log10(0.05)))
   698→    .attr('x2', width - margin.right)
   699→    .attr('y2', yScale(-Math.log10(0.05)))
   700→    .attr('stroke', 'blue')
   701→    .attr('stroke-dasharray', '5,5');
   702→    
   703→  // Axes
   704→  svg.append('g')
   705→    .attr('transform', \`translate(0, \${height - margin.bottom})\`)
   706→    .call(d3.axisBottom(xScale));
   707→    
   708→  svg.append('g')
   709→    .attr('transform', \`translate(\${margin.left}, 0)\`)
   710→    .call(d3.axisLeft(yScale));
   711→    
   712→  // Labels
   713→  svg.append('text')
   714→    .attr('x', width / 2)
   715→    .attr('y', height - 10)
   716→    .attr('text-anchor', 'middle')
   717→    .text('Log2 Fold Change');
   718→    
   719→  svg.append('text')
   720→    .attr('transform', 'rotate(-90)')
   721→    .attr('x', -(height / 2))
   722→    .attr('y', 20)
   723→    .attr('text-anchor', 'middle')
   724→    .text('-Log10(P-value)');
   725→}
