   566→const HighlightText = ({ text, highlight }: { text: string; highlight: string }) => {
   567→  if (!highlight.trim()) return <span>{text}</span>;
   568→  
   569→  const regex = new RegExp(`(${highlight.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
   570→  const parts = text.split(regex);
   571→  
   572→  return (
   573→    <span>
   574→      {parts.map((part, i) => 
   575→        regex.test(part) ? (
   576→          <strong key={i} className="bg-yellow-200 dark:bg-yellow-800 rounded px-0.5">{part}</strong>
   577→        ) : (
   578→          <span key={i}>{part}</span>
   579→        )
   580→      )}
   581→    </span>
   582→  );
