   564→const HighlightText = ({ text, highlight }: { text: string; highlight: string }) => {
   565→  if (!highlight.trim()) return <span>{text}</span>;
   566→  
   567→  const regex = new RegExp(`(${highlight.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
   568→  const parts = text.split(regex);
   569→  
   570→  return (
   571→    <span>
   572→      {parts.map((part, i) => 
   573→        regex.test(part) ? (
   574→          <strong key={i} className="bg-yellow-200 dark:bg-yellow-800 rounded px-0.5">{part}</strong>
   575→        ) : (
   576→          <span key={i}>{part}</span>
   577→        )
   578→      )}
   579→    </span>
   580→  );
