   663→function createVolcanoPlot(containerId) {
   664→  const container = document.getElementById(containerId);
   665→  
   666→  // Setup SVG
   667→  const width = 800;
   668→  const height = 500;
   669→  const margin = { top: 40, right: 40, bottom: 60, left: 70 };
   670→  
   671→  // Scales
   672→  const xScale = d3.scaleLinear()
   673→    .domain(d3.extent(data.genes, d => d.foldChange))
   674→    .range([margin.left, width - margin.right]);
   675→    
   676→  const yScale = d3.scaleLinear()
   677→    .domain([0, -Math.log10(d3.min(data.genes, d => d.pValue)) * 1.1])
   678→    .range([height - margin.bottom, margin.top]);
   679→  
   680→  // Draw points
   681→  const svg = d3.select(container).append('svg')
   682→    .attr('width', width)
   683→    .attr('height', height);
   684→    
   685→  svg.selectAll('circle')
   686→    .data(data.genes)
   687→    .enter()
   688→    .append('circle')
   689→    .attr('cx', d => xScale(d.foldChange))
   690→    .attr('cy', d => yScale(-Math.log10(d.pValue)))
   691→    .attr('r', d => d.significant ? 6 : 4)
   692→    .attr('fill', d => {
   693→      if (d.significant && d.foldChange > 0) return '#e41a1c'; // Red up
   694→      if (d.significant && d.foldChange < 0) return '#377eb8'; // Blue down
   695→      return '#999999'; // Gray not significant
   696→    })
   697→    .attr('opacity', 0.7)
   698→    .on('mouseover', showTooltip)
   699→    .on('mouseout', hideTooltip);
   700→    
   701→  // Threshold lines
   702→  svg.append('line')
   703→    .attr('x1', xScale(-1))
   704→    .attr('y1', margin.top)
   705→    .attr('x2', xScale(-1))
   706→    .attr('y2', height - margin.bottom)
   707→    .attr('stroke', 'green')
   708→    .attr('stroke-dasharray', '5,5');
   709→    
   710→  svg.append('line')
   711→    .attr('x1', xScale(1))
   712→    .attr('y1', margin.top)
   713→    .attr('x2', xScale(1))
   714→    .attr('y2', height - margin.bottom)
   715→    .attr('stroke', 'green')
   716→    .attr('stroke-dasharray', '5,5');
   717→    
   718→  svg.append('line')
   719→    .attr('x1', margin.left)
   720→    .attr('y1', yScale(-Math.log10(0.05)))
   721→    .attr('x2', width - margin.right)
   722→    .attr('y2', yScale(-Math.log10(0.05)))
   723→    .attr('stroke', 'blue')
   724→    .attr('stroke-dasharray', '5,5');
   725→    
   726→  // Axes
   727→  svg.append('g')
   728→    .attr('transform', \`translate(0, \${height - margin.bottom})\`)
   729→    .call(d3.axisBottom(xScale));
   730→    
   731→  svg.append('g')
   732→    .attr('transform', \`translate(\${margin.left}, 0)\`)
   733→    .call(d3.axisLeft(yScale));
   734→    
   735→  // Labels
   736→  svg.append('text')
   737→    .attr('x', width / 2)
   738→    .attr('y', height - 10)
   739→    .attr('text-anchor', 'middle')
   740→    .text('Log2 Fold Change');
   741→    
   742→  svg.append('text')
   743→    .attr('transform', 'rotate(-90)')
   744→    .attr('x', -(height / 2))
   745→    .attr('y', 20)
   746→    .attr('text-anchor', 'middle')
   747→    .text('-Log10(P-value)');
   748→}

  1189→const generateMarkdownOutput = (code: string): string => {
  1190→  const outputs: string[] = [];
  1191→  
  1192→  outputs.push('');
  1193→  outputs.push('#'.repeat(60));
  1194→  outputs.push('# RENDERED MARKDOWN PREVIEW');

  1200→  let inCodeBlock = false;
  1201→  
  1202→  for (const line of lines) {
  1203→    if (line.trim().startsWith('```')) {
  1204→      inCodeBlock = !inCodeBlock;
  1205→      if (inCodeBlock) {
  1206→        outputs.push('<div class="code-block">');
  1207→      } else {
  1208→        outputs.push('</div>');
  1209→      }
  1210→      continue;
  1211→    }
  1212→    
  1213→    if (inCodeBlock) {
  1214→      outputs.push('  ' + line);
  1215→      continue;
  1216→    }
  1217→    
  1218→    // Headers
  1219→    if (line.startsWith('# ')) {
  1220→      outputs.push(`<h1>${line.substring(2)}</h1>`);
  1221→    } else if (line.startsWith('## ')) {
  1222→      outputs.push(`<h2>${line.substring(3)}</h2>`);
  1223→    } else if (line.startsWith('### ')) {
  1224→      outputs.push(`<h3>${line.substring(4)}</h3>`);
  1225→    } else if (line.startsWith('> ')) {
  1226→      outputs.push(`<blockquote>${line.substring(2)}</blockquote>`);
  1227→    } else if (line.startsWith('- ') || line.startsWith('* ')) {
  1228→      outputs.push(`<li>${line.substring(2)}</li>`);
  1229→    } else if (line.match(/^\d+\.\s/)) {
  1230→      outputs.push(`<li>${line.replace(/^\d+\.\s/, '')}</li>`);
  1231→    } else if (line.trim() === '---') {
  1232→      outputs.push('<hr/>');
  1233→    } else if (line.trim()) {
  1234→      outputs.push(`<p>${line}</p>`);
  1235→    }
  1236→  }
  1237→  
  1238→  outputs.push('');
  1239→  outputs.push('#'.repeat(60));
  1240→  outputs.push('*Full Markdown rendering would include syntax highlighting,');
  1241→  outputs.push('table formatting, image embedding, and MathJax support*');
  1242→  outputs.push('#'.repeat(60));
  1243→  
  1244→  return outputs.join('\n');
