   570→   570→  return (

   574→   574→          <strong key={i} className="bg-yellow-200 dark:bg-yellow-800 rounded px-0.5">{part}</strong>
   575→   575→        ) : (
   576→   576→          <span key={i}>{part}</span>

   578→   578→      )}
