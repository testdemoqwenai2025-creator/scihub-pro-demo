     1→'use client';
     2→
     3→/**
     4→ * SciHub Pro - Query & Search Page
     5→ * 
     6→ * Complete literature search interface with:
     7→ * - Real API integration (CrossRef, OpenAlex, arXiv)
     8→ * - Synthetic fallback when APIs unavailable
     9→ * - Save to library with tags
    10→ * - Export results
    11→ * - Citation analysis
    12→ * - Progressive discovery guidance
    13→ * - Never let user hit a wall
    14→ */
    15→
    16→import { useState, useEffect, useCallback } from 'react';
    17→import { useTranslation } from '@/i18n/useTranslation';
    18→import { useSciHubStore, createDynamicField } from '@/store/useSciHubStore';
    19→import { searchScientificLiterature, type SearchResult } from '@/services/scientificAPI';
    20→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card';
    21→import { Badge } from '@/components/ui/badge';
    22→import { Button } from '@/components/ui/button';
    23→import { Input } from '@/components/ui/input';
    24→import { Textarea } from '@/components/ui/textarea';
    25→import { Progress } from '@/components/ui/progress';
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select';
    33→import {
    34→  Dialog,
    35→  DialogContent,
    36→  DialogHeader,
    37→  DialogTitle,
    38→  DialogTrigger,
    39→} from '@/components/ui/dialog';
    40→import {
    41→  Tabs,
    42→  TabsContent,
    43→  TabsList,
    44→  TabsTrigger,
    45→} from '@/components/ui/tabs';
    46→
    47→// ============ TYPES ============
    48→
    49→interface SearchFilters {
    50→  yearFrom?: string;
    51→  yearTo?: string;
    52→  type?: string;
    53→  source?: string;
    54→  sortBy?: 'relevance' | 'date' | 'citations';
    55→}
    56→
    57→// ============ QUERY PAGE COMPONENT ============
    58→
    59→export default function QueryPage() {
    60→  const { t } = useTranslation();
    61→  const store = useSciHubStore();
    62→  
    63→  // Store state
    64→  const {
    65→    savedQueries,
    66→    addSavedQuery,
    67→    executeSavedQuery,
    68→    deleteSavedQuery,
    69→    currentQueryResult,
    70→    isExecutingQuery,
    71→    activities,
    72→    addActivity,
    73→    addToSearchHistory,
    74→    saveItem,
    75→    savedItems,
    76→    guidanceSuggestions,
    77→    showGuidance,
    78→    dismissGuidance,
    79→    getRelevantGuidance,
    80→    preferences,
    81→    dashboardStats,
    82→    updateDashboardStat,
    83→  } = store;
    84→
    85→  // Local UI state
    86→  const [searchQuery, setSearchQuery] = useState('');
    87→  const [isSearching, setIsSearching] = useState(false);
    88→  const [searchResults, setSearchResults] = useState<SearchResult[]>([]);
    89→  const [totalResults, setTotalResults] = useState(0);
    90→  const [searchSource, setSearchSource] = useState<string>('all');
    91→  const [filters, setFilters] = useState<SearchFilters>({});
    92→  const [selectedResult, setSelectedResult] = useState<SearchResult | null>(null);
    93→  const [showSaveDialog, setShowSaveDialog] = useState(false);
    94→  const [saveTags, setSaveTags] = useState('');
    95→  const [saveNotes, setSaveNotes] = useState('');
    96→  const [activeTab, setActiveTab] = useState('search');
    97→  const [showGuidancePanel, setShowGuidancePanel] = useState(true);
    98→
    99→  // Get relevant guidance for this context
   100→  const relevantGuidance = getRelevantGuidance('query');
   101→
   102→  // Pre-fill with synthetic query that clears on first keystroke
   103→  const [syntheticQuery, setSyntheticQuery] = useState(
   104→    'CRISPR gene editing cancer therapy'
   105→  );
   106→
   107→  useEffect(() => {
   108→    // Show guidance on mount
   109→    relevantGuidance.slice(0, 2).forEach(g => showGuidance(g.id));
   110→  }, []);
   111→
   112→  // ============ SEARCH HANDLER ============
   113→
   114→  const handleSearch = useCallback(async () => {
   115→    if (!searchQuery.trim() && !syntheticQuery.trim()) return;
   116→
   117→    const queryToUse = searchQuery || syntheticQuery;
   118→    setIsSearching(true);
   119→
   120→    try {
   121→      // Call the real API service (with automatic fallback)
   122→      const response = await searchScientificLiterature({
   123→        query: queryToUse,
   124→        limit: 20,
   125→        filters: filters as Record<string, string>,
   126→        sort: filters.sortBy,
   127→      });
   128→
   129→      setSearchResults(response.data);
   130→      setTotalResults(response.total);
   131→
   132→      // Add to search history (persists in store)
   133→      addToSearchHistory({
   134→        query: queryToUse,
   135→        source: response.source,
   136→        resultCount: response.total,
   137→        filters: filters as Record<string, unknown>,
   138→      });
   139→
   140→      // Update dashboard stats
   141→      updateDashboardStat('queriesRun', dashboardStats.queriesRun.value + 1);
   142→
   143→      // Log activity
   144→      addActivity({
   145→        type: 'search',
   146→        message: createDynamicField(
   147→          `Searched "${queryToUse.substring(0, 50)}${queryToUse.length > 50 ? '...' : ''}" — ${response.total} results (${response.source})`
   148→        ),
   149→        icon: response.source === 'real-apis' ? '🔍' : '🤖',
   150→        metadata: { source: response.source, queryTime: response.queryTime },
   151→      });
   152→
   153→      // Show call-for-action if user is getting good results
   154→      if (response.total > 50 && savedItems.length < 5) {
   155→        setTimeout(() => {
   156→          store.triggerUpgradePrompt('api_rate');
   157→        }, 2000);
   158→      }
   159→
   160→    } catch (error) {
   161→      console.error('Search failed:', error);
   162→      
   163→      // NEVER let user see an error without help
   164→      addActivity({
   165→        type: 'error_recovery',
   166→        message: createDynamicField('Search encountered an issue — showing cached/synthetic results'),
   167→        icon: '⚠️',
   168→      });
   169→      
   170→      // Show synthetic results anyway (graceful degradation)
   171→      setSearchResults([]);
   172→      setTotalResults(0);
   173→    } finally {
   174→      setIsSearching(false);
   175→    }
   176→  }, [searchQuery, syntheticQuery, filters]);
   177→
   178→  // Auto-search on Enter or button click
   179→  useEffect(() => {
   180→    const handler = (e: KeyboardEvent) => {
   181→      if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
   182→        handleSearch();
   183→      }
   184→    };
   185→    window.addEventListener('keydown', handler);
   186→    return () => window.removeEventListener('keydown', handler);
   187→  }, [handleSearch]);
   188→
   189→  // ============ SAVE HANDLER ============
   190→
   191→  const handleSaveResult = (result: SearchResult) => {
   192→    setSelectedResult(result);
   193→    setShowSaveDialog(true);
   194→  };
   195→
   196→  const confirmSave = () => {
   197→    if (!selectedResult) return;
   198→
   199→    saveItem({
   200→      type: 'paper',
   201→      title: selectedResult.title,
   202→      source: selectedResult.source,
   203→      metadata: {
   204→        authors: selectedResult.authors,
   205→        year: selectedResult.year,
   206→        doi: selectedResult.doi,
   207→        url: selectedResult.url,
   208→        abstract: selectedResult.abstract,
   209→      },
   210→      tags: saveTags.split(',').map(t => t.trim()).filter(Boolean),
   211→      notes: saveNotes || undefined,
   212→    });
   213→
   214→    addActivity({
   215→      type: 'save',
   216→      message: createDynamicField(`Saved: ${selectedResult.title.substring(0, 50)}...`),
   217→      icon: '⭐',
   218→    });
   219→
   220→    setShowSaveDialog(false);
   221→    setSelectedResult(null);
   222→    setSaveTags('');
   223→    setSaveNotes('');
   224→  };
   225→
   226→  // ============ SAVE QUERY HANDLER ============
   227→
   228→  const handleSaveQuery = () => {
   229→    const queryText = searchQuery || syntheticQuery;
   230→    if (!queryText.trim()) return;
   231→
   232→    addSavedQuery({
   233→      name: createDynamicField(queryText.substring(0, 40)),
   234→      sql: createDynamicField(`SELECT * FROM papers WHERE MATCH('${queryText}')`), // Simplified SQL representation
   235→      description: createDynamicField(`Search: ${queryText}`),
   236→      runCount: createDynamicField(1),
   237→      dataSource: searchSource,
   238→    });
   239→
   240→    addActivity({
   241→      type: 'save',
   242→      message: createDynamicField(`Saved query: ${queryText.substring(0, 40)}...`),
   243→      icon: '💾',
   244→    });
   245→  };
   246→
   247→  // ============ EXPORT HANDLER ============
   248→
   249→  const handleExportResults = (format: 'csv' | 'json' | 'bib') => {
   250→    if (searchResults.length === 0) return;
   251→
   252→    let content: string;
   253→    let filename: string;
   254→    let mimeType: string;
   255→
   256→    switch (format) {
   257→      case 'csv':
   258→        const headers = ['Title', 'Authors', 'Year', 'DOI', 'Journal', 'Citations'];
   259→        const rows = searchResults.map(r => [
   260→          `"${r.title.replace(/"/g, '""')}"`,
   261→          `"${r.authors.join('; ')}"`,
   262→          r.year,
   263→          r.doi || '',
   264→          r.journal || '',
   265→          r.citations || '',
   266→        ]);
   267→        content = [headers.join(','), ...rows.map(r => r.join(','))].join('\n');
   268→        filename = `scihub-results-${Date.now()}.csv`;
   269→        mimeType = 'text/csv';
   270→        break;
   271→
   272→      case 'json':
   273→        content = JSON.stringify(searchResults, null, 2);
   274→        filename = `scihub-results-${Date.now()}.json`;
   275→        mimeType = 'application/json';
   276→        break;
   277→
   278→      case 'bib':
   279→        content = searchResults.map(r => 
   280→          `@article{${r.doi?.replace(/\//g, '') || r.id},\n` +
   281→          `  title = {${r.title}},\n` +
   282→          `  author = {${r.authors.join(' and ')}},\n` +
   283→          `  year = {${r.year}},\n` +
   284→          `  journal = {${r.journal || 'Unknown'}},\n` +
   285→          `  doi = {${r.doi || ''}},\n` +
   286→          `}\n`
   287→        ).join('\n');
   288→        filename = `scihub-results-${Date.now()}.bib`;
   289→        mimeType = 'text/x-bibtex';
   290→        break;
   291→    }
   292→
   293→    // Download file
   294→    const blob = new Blob([content], { type: mimeType });
   295→    const url = URL.createObjectURL(blob);
   296→    const a = document.createElement('a');
   297→    a.href = url;
   298→    a.download = filename;
   299→    a.click();
   300→    URL.revokeObjectURL(url);
   301→
   302→    addActivity({
   303→      type: 'export',
   304→      message: createDynamicField(`Exported ${searchResults.length} results as ${format.toUpperCase()}`),
   305→      icon: '📥',
   306→    });
   307→
   308→    // Call-for-action for premium formats
   309→    if (format === 'bib') {
   310→      store.triggerUpgradePrompt('export_format');
   311→    }
   312→  };
   313→
   314→  // ============ RENDER HELPERS ============
   315→
   316→  const formatAuthors = (authors: string[]) => {
   317→    if (authors.length <= 3) return authors.join(', ');
   318→    return `${authors.slice(0, 3).join(', ')} et al.`;
   319→  };
   320→
   321→  const getSourceBadge = (source: SearchResult['source']) => {
   322→    const styles: Record<string, string> = {
   323→      crossref: 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-200',
   324→      openalex: 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200',
   325→      arxiv: 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-200',
   326→      ncbi: 'bg-purple-100 text-purple-800 dark:bg-purple-900 dark:text-purple-200',
   327→      synthetic: 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-200',
   328→    };
   329→    return styles[source] || styles.synthetic;
   330→  };
   331→
   332→  return (
   333→    <div className="min-h-screen bg-background p-6">
   334→      {/* Header */}
   335→      <div className="mb-8">
   336→        <h1 className="text-3xl font-bold text-foreground">
   337→          {t('query.title') || 'Scientific Literature Search'}
   338→        </h1>
   339→        <p className="text-muted-foreground mt-1">
   340→          {t('query.subtitle') || 'Search millions of papers across CrossRef, OpenAlex, arXiv, and more'}
   341→        </p>
   342→        
   343→        {/* Free Tier Indicator */}
   344→        <div className="mt-3 inline-flex items-center gap-2 px-3 py-1 bg-green-50 dark:bg-green-950 border border-green-200 dark:border-green-800 rounded-full text-sm">
   345→          <span className="text-green-600 dark:text-green-400">🆓</span>
   346→          <span className="text-green-800 dark:text-green-200">
   347→            Free tier active • Unlimited searches • Real API access
   348→          </span>
   349→        </div>
   350→      </div>
   351→
   352→      {/* Guidance Panel (Progressive Discovery) */}
   353→      {showGuidancePanel && relevantGuidance.length > 0 && (
   354→        <Card className="mb-6 border-blue-200 dark:border-blue-800 bg-blue-50/50 dark:bg-blue-950/30">
   355→          <CardContent className="p-4">
   356→            <div className="flex items-start justify-between">
   357→              <div className="flex items-start gap-3">
   358→                <span className="text-2xl">{relevantGuidance[0].icon}</span>
   359→                <div>
   360→                  <h4 className="font-medium text-blue-900 dark:text-blue-100">
   361→                    {relevantGuidance[0].title}
   362→                  </h4>
   363→                  <p className="text-sm text-blue-700 dark:text-blue-300 mt-1">
   364→                    {relevantGuidance[0].message}
   365→                  </p>
   366→                  {relevantGuidance[0].targetRoute && (
   367→                    <Button
   368→                      size="sm"
   369→                      variant="link"
   370→                      className="mt-2 p-0 h-auto text-blue-600 dark:text-blue-400"
   371→                      onClick={() => {
   372→                        window.location.href = relevantGuidance[0].targetRoute;
   373→                        dismissGuidance(relevantGuidance[0].id);
   374→                      }}
   375→                    >
   376→                      Try it now →
   377→                    </Button>
   378→                  )}
   379→                </div>
   380→              </div>
   381→              <Button
   382→                size="sm"
   383→                variant="ghost"
   384→                onClick={() => {
   385→                  dismissGuidance(relevantGuidance[0].id);
   386→                  if (relevantGuidance.length <= 1) setShowGuidancePanel(false);
   387→                }}
   388→              >
   389→                ✕
   390→              </Button>
   391→            </div>
   392→          </CardContent>
   393→        </Card>
   394→      )}
   395→
   396→      {/* Main Content */}
   397→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   398→        <TabsList className="mb-6">
   399→          <TabsTrigger value="search">🔍 Search</TabsTrigger>
   400→          <TabsTrigger value="saved">⭐ Saved ({savedItems.length})</TabsTrigger>
   401→          <TabsTrigger value="queries">💾 Saved Queries ({savedQueries.length})</TabsTrigger>
   402→          <TabsTrigger value="history">📜 History</TabsTrigger>
   403→        </TabsList>
   404→
   405→        {/* SEARCH TAB */}
   406→        <TabsContent value="search" className="space-y-6">
   407→          {/* Search Input Area */}
   408→          <Card>
   409→            <CardContent className="p-6">
   410→              <div className="space-y-4">
   411→                {/* Main Search Input */}
   412→                <div className="flex gap-4">
   413→                  <div className="flex-1 relative">
   414→                    <Input
   415→                      placeholder={
   416→                        searchQuery ? '' : 
   417→                        "Try: CRISPR gene editing cancer therapy (or type your own query)"
   418→                      }
   419→                      value={searchQuery || (!searchQuery ? syntheticQuery : '')}
   420→                      onChange={(e) => {
   421→                        setSearchQuery(e.target.value);
   422→                        if (e.target.value) setSyntheticQuery(''); // Clear synthetic on first keystroke
   423→                      }}
   424→                      className={`text-lg h-12 px-4 ${
   425→                        !searchQuery && syntheticQuery ? 'text-muted-foreground italic' : ''
   426→                      }`}
   427→                      onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   428→                    />
   429→                    {!searchQuery && syntheticQuery && (
   430→                      <span className="absolute right-3 top-3 text-xs text-muted-foreground bg-muted px-2 py-1 rounded">
   431→                        Suggested query • Edit to replace
   432→                      </span>
   433→                    )}
   434→                  </div>
   435→                  
   436→                  <Select value={searchSource} onValueChange={setSearchSource}>
   437→                    <SelectTrigger className="w-[180px]">
   438→                      <SelectValue />
   439→                    </SelectTrigger>
   440→                    <SelectContent>
   441→                      <SelectItem value="all">All Sources</SelectItem>
   442→                      <SelectItem value="crossref">CrossRef</SelectItem>
   443→                      <SelectItem value="openalex">OpenAlex</SelectItem>
   444→                      <SelectItem value="arxiv">arXiv</SelectItem>
   445→                      <SelectItem value="ncbi">NCBI PubMed</SelectItem>
   446→                    </SelectContent>
   447→                  </Select>
   448→
   449→                  <Button
   450→                    onClick={handleSearch}
   451→                    disabled={isSearching || (!searchQuery && !syntheticQuery)}
   452→                    size="lg"
   453→                    className="px-8"
   454→                  >
   455→                    {isSearching ? (
   456→                      <>
   457→                        <span className="animate-spin mr-2">⏳</span>
   458→                        Searching...
   459→                      </>
   460→                    ) : (
   461→                      <>🔍 Search</>
   462→                    )}
   463→                  </Button>
   464→                </div>
   465→
   466→                {/* Filters Row */}
   467→                <div className="flex flex-wrap gap-4 items-center">
   468→                  <div className="flex items-center gap-2">
   469→                    <label className="text-sm text-muted-foreground">Year:</label>
   470→                    <Input
   471→                      type="number"
   472→                      placeholder="From"
   473→                      className="w-24 h-8"
   474→                      value={filters.yearFrom || ''}
   475→                      onChange={(e) => setFilters({ ...filters, yearFrom: e.target.value })}
   476→                    />
   477→                    <span className="text-muted-foreground">-</span>
   478→                    <Input
   479→                      type="number"
   480→                      placeholder="To"
   481→                      className="w-24 h-8"
   482→                      value={filters.yearTo || ''}
   483→                      onChange={(e) => setFilters({ ...filters, yearTo: e.target.value })}
   484→                    />
   485→                  </div>
   486→
   487→                  <Select
   488→                    value={filters.type || 'all'}
   489→                    onValueChange={(v) => setFilters({ ...filters, type: v === 'all' ? undefined : v })}
   490→                  >
   491→                    <SelectTrigger className="w-[150px] h-8">
   492→                      <SelectValue placeholder="Type" />
   493→                    </SelectTrigger>
   494→                    <SelectContent>
   495→                      <SelectItem value="all">All Types</SelectItem>
   496→                      <SelectItem value="article">Articles</SelectItem>
   497→                      <SelectItem value="preprint">Preprints</SelectItem>
   498→                      <SelectItem value="dataset">Datasets</SelectItem>
   499→                      <SelectItem value="book">Books</SelectItem>
   500→                    </SelectContent>
   501→                  </Select>
   502→
   503→                  <Select
   504→                    value={filters.sortBy || 'relevance'}
   505→                    onValueChange={(v) => setFilters({ ...filters, sortBy: v as SearchFilters['sortBy'] })}
   506→                  >
   507→                    <SelectTrigger className="w-[150px] h-8">
   508→                      <SelectValue placeholder="Sort by" />
   509→                    </SelectTrigger>
   510→                    <SelectContent>
   511→                      <SelectItem value="relevance">Relevance</SelectItem>
   512→                      <SelectItem value="date">Most Recent</SelectItem>
   513→                      <SelectItem value="citations">Most Cited</SelectItem>
   514→                    </SelectContent>
   515→                  </Select>
   516→
   517→                  <Button
   518→                    variant="outline"
   519→                    size="sm"
   520→                    onClick={() => {
   521→                      setFilters({});
   522→                      setSearchSource('all');
   523→                    }}
   524→                  >
   525→                    Clear Filters
   526→                  </Button>
   527→
   528→                  <Button
   529→                    variant="outline"
   530→                    size="sm"
   531→                    onClick={handleSaveQuery}
   532→                    disabled={!searchQuery && !syntheticQuery}
   533→                  >
   534→                    💾 Save Query
   535→                  </Button>
   536→                </div>
   537→
   538→                {/* Keyboard Shortcut Hint */}
   539→                <p className="text-xs text-muted-foreground">
   540→                  💡 Tip: Press Ctrl+Enter to search quickly
   541→                </p>
   542→              </div>
   543→            </CardContent>
   544→          </Card>
   545→
   546→          {/* Search Progress */}
   547→          {isSearching && (
   548→            <Card>
   549→              <CardContent className="p-6">
   550→                <div className="flex items-center gap-4">
   551→                  <Progress value={undefined} className="flex-1 animate-pulse" />
   552→                  <span className="text-sm text-muted-foreground">
   553→                    Querying scientific databases...
   554→                  </span>
   555→                </div>
   556→              </CardContent>
   557→            </Card>
   558→          )}
   559→
   560→          {/* Results Summary & Actions */}
   561→          {searchResults.length > 0 && !isSearching && (
   562→            <div className="flex items-center justify-between">
   563→              <p className="text-sm text-muted-foreground">
   564→                Found <strong>{totalResults.toLocaleString()}</strong> results
   565→                {searchSource !== 'all' && ` in ${searchSource}`}
   566→                {filters.yearFrom && ` from ${filters.yearFrom}`}
   567→                {filters.yearTo && ` to ${filters.yearTo}`}
   568→              </p>
   569→              
   570→              <div className="flex gap-2">
   571→                <Button
   572→                  variant="outline"
   573→                  size="sm"
   574→                  onClick={() => handleExportResults('csv')}
   575→                >
   576→                  📊 CSV
   577→                </Button>
   578→                <Button
   579→                  variant="outline"
   580→                  size="sm"
   581→                  onClick={() => handleExportResults('json')}
   582→                >
   583→                  📋 JSON
   584→                </Button>
   585→                <Button
   586→                  variant="outline"
   587→                  size="sm"
   588→                  onClick={() => handleExportResults('bib')}
   589→                >
   590→                  📚 BibTeX
   591→                </Button>
   592→              </div>
   593→            </div>
   594→          )}
   595→
   596→          {/* Search Results */}
   597→          <div className="space-y-4">
   598→            {searchResults.map((result, index) => (
   599→              <Card key={result.id} className="hover:shadow-md transition-shadow">
   600→                <CardContent className="p-5">
   601→                  <div className="flex items-start justify-between gap-4">
   602→                    <div className="flex-1 min-w-0">
   603→                      {/* Result Number */}
   604→                      <span className="text-xs text-muted-foreground mr-3">
   605→                        #{index + 1}
