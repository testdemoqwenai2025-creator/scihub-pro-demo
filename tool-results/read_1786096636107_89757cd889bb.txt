   166→export default function ComputePage() {
   167→  const { t } = useTranslation();
   168→  const {
   169→    createJob,
   170→    updateJobProgress,
   171→    cancelJob,
   172→    addActivity,
   173→  } = useDynamicStore();
   174→
   175→  // UI State
   176→  const [jobs, setJobs] = useState<ComputeJob[]>(INITIAL_JOBS);
   177→  const [nodes] = useState<ComputeNode[]>(INITIAL_NODES);

   179→  const [showCreateForm, setShowCreateForm] = useState(false);
   180→  
   181→  // Form state
   182→  const [newJobName, setNewJobName] = useState('');
   183→  const [newJobType, setNewJobType] = useState<ComputeJob['type']>('analysis');
   184→  const [newJobPriority, setNewJobPriority] = useState<ComputeJob['priority']>('normal');
   185→  const [newJobDataset, setNewJobDataset] = useState('');
   186→  const [newJobGpus, setNewJobGpus] = useState(1);
   187→
   188→  // Auto-update running jobs
   189→  useEffect(() => {
   190→    const interval = setInterval(() => {
   191→      setJobs(prev => prev.map(job => {
   192→        if (job.status === 'running') {
   193→          const newProgress = Math.min(100, job.progress + Math.random() * 2);
   194→          const newComputeHours = job.computeHoursUsed + (Math.random() * 0.1);
   195→          
   196→          return {
   197→            ...job,
   198→            progress: newProgress,
   199→            computeHoursUsed: newComputeHours,
   200→            ...(newProgress >= 100 ? { 
   201→              status: 'completed' as const, 
   202→              completedAt: new Date(),
   203→              estimatedCost: job.estimatedCost ? job.estimatedCost + Math.random() * 0.5 : undefined
   204→            } : {}),
   205→          };
   206→        }
   207→        return job;
   208→      }));
   209→    }, 3000);
   210→
   211→    return () => clearInterval(interval);
   212→  }, []);
   213→
   214→  // Stats calculations
   215→  const activeJobs = jobs.filter(j => j.status === 'running');
   216→  const totalCompute = jobs.reduce((acc, j) => acc + j.computeHoursUsed, 0);
   217→  const totalGPUs = activeJobs.reduce((acc, j) => acc + j.gpusAllocated, 0);
   218→  const avgProgress = activeJobs.length > 0 
   219→    ? Math.round(activeJobs.reduce((acc, j) => acc + j.progress, 0) / activeJobs.length)
   220→    : 0;
   221→
   222→  // Handle job creation
   223→  const handleCreateJob = () => {
   224→    if (!newJobName.trim()) return;
   225→
   226→    const costPerHour = newJobGpus * 0.5; // $0.50 per GPU hour (free tier simulation)
   227→    const estimatedHours = newJobType === 'training' ? 24 : newJobType === 'simulation' ? 16 : 4;
   228→
   229→    const newJob: ComputeJob = {
   230→      id: `job-${Date.now()}`,
   231→      name: newJobName,
   232→      type: newJobType,
   233→      status: 'queued',
   234→      priority: newJobPriority,
   235→      submitter: 'You',
   236→      progress: 0,
   237→      computeHoursUsed: 0,
   238→      computeHoursTotal: estimatedHours,
   239→      gpusAllocated: 0,
   240→      memoryUsed: '0 GB',
   241→      memoryTotal: `${newJobGpus * 16} GB`,
   242→      submittedAt: new Date(),
   243→      estimatedCost: estimatedHours * costPerHour,
   244→      logs: [{ timestamp: new Date(), level: 'INFO', message: 'Job created and queued' }],
   245→    };
   246→
   247→    setJobs(prev => [newJob, ...prev]);
   248→    
   249→    // Also add to store
   250→    createJob({
   251→      name: createDynamicField(newJobName),
   252→      type: createDynamicField(newJobType),
   253→      status: createDynamicField('queued'),
   254→      priority: createDynamicField(newJobPriority),
   255→      submitter: createDynamicField('You'),
   256→      config: {
   257→        inputDataset: createDynamicField(newJobDataset),
   258→        parameters: {},
   259→        outputFormat: createDynamicField('JSON'),
   260→        notifications: createDynamicField(true),
   261→        retryCount: createDynamicField(3),
   262→      },
   263→      progress: createDynamicField(0),
   264→      computeHoursUsed: createDynamicField(0),
   265→      computeHoursTotal: createDynamicField(estimatedHours),
   266→      gpusAllocated: createDynamicField(0),
   267→      memoryUsed: createDynamicField('0 GB'),
   268→    });
   269→
   270→    addActivity({
   271→      type: 'job',
   272→      message: createDynamicField(`Submitted job "${newJobName}"`),
   273→      icon: '⚡',
   274→    });
   275→
   276→    // Reset form
   277→    setNewJobName('');
   278→    setShowCreateForm(false);
   279→  };
   280→
   281→  // Handle job cancellation
   282→  const handleCancelJob = (jobId: string) => {
   283→    const job = jobs.find(j => j.id === jobId);
   284→    if (!job) return;
   285→
   286→    if (confirm(`Cancel job "${job.name}"?`)) {
   287→      setJobs(prev => prev.map(j => 
   288→        j.id === jobId ? { ...j, status: 'cancelled' as const } : j
   289→      ));
   290→      cancelJob(jobId);
   291→      
   292→      addActivity({
   293→        type: 'update',
   294→        message: createDynamicField(`Cancelled job "${job.name}"`),
   295→        icon: '🛑',
   296→      });
   297→    }
   298→  };
   299→
   300→  // Get status badge
   301→  const getStatusBadge = (status: ComputeJob['status']) => {
   302→    switch (status) {
   303→      case 'running': return <Badge className="bg-blue-500 animate-pulse">{t('compute.running') || 'Running'}</Badge>;
   304→      case 'queued': return <Badge className="bg-yellow-500">{t('compute.queued') || 'Queued'}</Badge>;
   305→      case 'completed': return <Badge className="bg-green-500">{t('compute.completed') || 'Completed'}</Badge>;
   306→      case 'failed': return <Badge variant="destructive">{t('compute.failed') || 'Failed'}</Badge>;
   307→      case 'cancelled': return <Badge variant="secondary">Cancelled</Badge>;
   308→      default: return <Badge variant="secondary">{status}</Badge>;
   309→    }
   310→  };
   311→
   312→  // Get priority badge
   313→  const getPriorityBadge = (priority: ComputeJob['priority']) => {
   314→    const colors: Record<string, string> = {
   315→      low: 'bg-gray-400',
   316→      normal: 'bg-blue-400',
   317→      high: 'bg-orange-400',
   318→      urgent: 'bg-red-500',
   319→    };
   320→    return <Badge className={colors[priority]}>{priority.charAt(0).toUpperCase() + priority.slice(1)}</Badge>;
   321→  };
   322→
   323→  // Format time ago
   324→  const formatTimeAgo = (date: Date): string => {
   325→    const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   326→    if (seconds < 60) return `${seconds}s ago`;
   327→    const minutes = Math.floor(seconds / 60);
   328→    if (minutes < 60) return `${minutes}m ago`;
   329→    const hours = Math.floor(minutes / 60);
   330→    if (hours < 24) return `${hours}h ago`;
   331→    return `${Math.floor(hours / 24)}d ago`;
   332→  };
   333→
   334→  return (
   335→    <div className="min-h-screen bg-background p-6">
   336→      {/* Header */}
   337→      <div className="mb-6">
   338→        <h1 className="text-3xl font-bold text-foreground">{t('compute.title') || 'Compute Layer'}</h1>
   339→        <p className="text-muted-foreground mt-1">
   340→          Manage computational jobs and monitor cluster resources. Free-tier simulation included.
   341→        </p>
   342→        
   343→        <div className="mt-2 flex items-center gap-3">
   344→          <Badge variant="secondary" className="bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200">
   345→            🆓 Free Tier Compute
   346→          </Badge>
   347→          <span className="text-sm text-muted-foreground">
   348→            Simulated resources • No actual costs incurred
   349→          </span>
   350→        </div>
   351→      </div>
   352→
   353→      {/* Stats Grid */}
   354→      <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-6">
   355→        <Card>
   356→          <CardContent className="p-4 flex items-center gap-3">
   357→            <span className="text-2xl">⚡</span>
   358→            <div>
   359→              <p className="text-sm text-muted-foreground">{t('compute.active_jobs') || 'Active Jobs'}</p>
   360→              <p className="text-xl font-bold text-blue-500">{activeJobs.length}</p>
   361→            </div>
   362→          </CardContent>
   363→        </Card>
   364→        <Card>
   365→          <CardContent className="p-4 flex items-center gap-3">
   366→            <span className="text-2xl">⏱️</span>
   367→            <div>
   368→              <p className="text-sm text-muted-foreground">Total Compute Used</p>
   369→              <p className="text-xl font-bold">{totalCompute.toFixed(1)} hrs</p>
   370→            </div>
   371→          </CardContent>
   372→        </Card>
   373→        <Card>
   374→          <CardContent className="p-4 flex items-center gap-3">
   375→            <span className="text-2xl">🖥️</span>
   376→            <div>
   377→              <p className="text-sm text-muted-foreground">GPUs Allocated</p>
   378→              <p className="text-xl font-bold text-purple-500">{totalGPUs}</p>
   379→            </div>
   380→          </CardContent>
   381→        </Card>
   382→        <Card>
   383→          <CardContent className="p-4 flex items-center gap-3">
   384→            <span className="text-2xl">📊</span>
   385→            <div>
   386→              <p className="text-sm text-muted-foreground">Avg Progress</p>
   387→              <p className="text-xl font-bold text-green-500">{avgProgress}%</p>
   388→            </div>
   389→          </CardContent>
   390→        </Card>
   391→      </div>
   392→
   393→      {/* Create Job Button */}
   394→      <div className="mb-6 flex justify-between items-center">
   395→        <h2 className="text-xl font-semibold">{t('compute.jobs') || 'Jobs'}</h2>
   396→        <Button onClick={() => setShowCreateForm(!showCreateForm)}>
   397→          + {t('compute.new_job') || 'New Job'}
   398→        </Button>
   399→      </div>
   400→
   401→      {/* Create Job Form */}
   402→      {showCreateForm && (
   403→        <Card className="mb-6 border-primary">
   404→          <CardHeader>
   405→            <CardTitle className="text-base">Submit New Job</CardTitle>
   406→          </CardHeader>
   407→          <CardContent>
   408→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-4">
   409→              <div>
   410→                <label className="text-sm font-medium">Job Name *</label>
   411→                <Input
   412→                  value={newJobName}
   413→                  onChange={(e) => setNewJobName(e.target.value)}
   414→                  placeholder="e.g., Variant Analysis Pipeline"
   415→                  className="mt-1"
   416→                />
   417→              </div>
   418→              <div>
   419→                <label className="text-sm font-medium">Job Type</label>
   420→                <Select value={newJobType} onValueChange={(v) => setNewJobType(v as ComputeJob['type'])}>
   421→                  <SelectTrigger className="mt-1">
   422→                    <SelectValue />
   423→                  </SelectTrigger>
   424→                  <SelectContent>
   425→                    <SelectItem value="analysis">Analysis</SelectItem>
   426→                    <SelectItem value="training">Training</SelectItem>
   427→                    <SelectItem value="simulation">Simulation</SelectItem>
   428→                    <SelectItem value="pipeline">Pipeline</SelectItem>
   429→                    <SelectItem value="custom">Custom</SelectItem>
   430→                  </SelectContent>
   431→                </Select>
   432→              </div>
   433→              <div>
   434→                <label className="text-sm font-medium">Priority</label>
   435→                <Select value={newJobPriority} onValueChange={(v) => setNewJobPriority(v as ComputeJob['priority'])}>
   436→                  <SelectTrigger className="mt-1">
   437→                    <SelectValue />
   438→                  </SelectTrigger>
   439→                  <SelectContent>
   440→                    <SelectItem value="low">Low</SelectItem>
   441→                    <SelectItem value="normal">Normal</SelectItem>
   442→                    <SelectItem value="high">High</SelectItem>
   443→                    <SelectItem value="urgent">Urgent</SelectItem>
   444→                  </SelectContent>
   445→                </Select>
   446→              </div>
   447→              <div>
   448→                <label className="text-sm font-medium">GPU Count</label>
   449→                <Select value={String(newJobGpus)} onValueChange={(v) => setNewJobGpus(parseInt(v))}>
   450→                  <SelectTrigger className="mt-1">
   451→                    <SelectValue />
   452→                  </SelectTrigger>
   453→                  <SelectContent>
   454→                    <SelectItem value="0">0 (CPU only)</SelectItem>
   455→                    <SelectItem value="1">1 GPU</SelectItem>
   456→                    <SelectItem value="2">2 GPUs</SelectItem>
   457→                    <SelectItem value="4">4 GPUs</SelectItem>
   458→                    <SelectItem value="8">8 GPUs</SelectItem>
   459→                  </SelectContent>
   460→                </Select>
   461→              </div>
   462→              <div className="md:col-span-2">
   463→                <label className="text-sm font-medium">Input Dataset (optional)</label>
   464→                <Input
   465→                  value={newJobDataset}
   466→                  onChange={(e) => setNewJobDataset(e.target.value)}
   467→                  placeholder="Dataset name or path..."
   468→                  className="mt-1"
   469→                />
   470→              </div>
   471→            </div>
   472→
   473→            {/* Cost Estimate */}
   474→            <div className="p-3 rounded-lg bg-muted/50 mb-4">
   475→              <div className="flex justify-between text-sm">
   476→                <span>Estimated Cost:</span>
   477→                <span className="font-medium text-green-600">$0.00 (Free Tier)</span>
   478→              </div>

   480→                In production, this would be ${(newJobGpus * 0.5 * (newJobType === 'training' ? 24 : 4)).toFixed(2)}

   482→            </div>
   483→
   484→            <div className="flex justify-end gap-3">
   485→              <Button variant="outline" onClick={() => setShowCreateForm(false)}>

   491→            </div>
   492→          </CardContent>
   493→        </Card>

   495→
   496→      <div className="flex gap-6">
   497→        {/* Jobs List */}
   498→        <div className="flex-1 space-y-4">
   499→          {jobs.map(job => (
   500→            <Card 
   501→              key={job.id} 
   502→              className={`cursor-pointer transition-all hover:shadow-md ${
   503→                selectedJob?.id === job.id ? 'ring-2 ring-primary' : ''
   504→              }`}
   505→              onClick={() => setSelectedJob(job)}
   506→            >
   507→              <CardContent className="p-4">
   508→                <div className="flex items-start justify-between mb-3">
   509→                  <div>
   510→                    <h3 className="font-medium">{job.name}</h3>
   511→                    <p className="text-sm text-muted-foreground">
   512→                      {job.submitter} • Submitted {formatTimeAgo(job.submittedAt)}
   513→                    </p>
   514→                  </div>
   515→                  <div className="flex gap-2">
   516→                    {getStatusBadge(job.status)}
   517→                    {getPriorityBadge(job.priority)}
   518→                  </div>
   519→                </div>
   520→
   521→                {job.status === 'running' && (
   522→                  <div className="space-y-2">

   611→                        <span className="text-muted-foreground">Completed</span>

   615→                    {selectedJob.estimatedCost !== undefined && (

   617→                        <span className="text-muted-foreground">Est. Cost</span>

   623→                  <div className="pt-2 border-t space-y-2">

   646→                    {selectedJob.logs.slice(-10).map((log, idx) => (

   649→                        log.level === 'WARN' ? 'text-yellow-500' :
   650→                        log.level === 'SUCCESS' ? 'text-green-500' :
   651→                        log.level === 'DEBUG' ? 'text-gray-400' :

   654→                        <span className="text-muted-foreground mr-2">
   655→                          [{log.timestamp.toLocaleTimeString()}]

   657→                        [{log.level}] {log.message}
   658→                      </div>
   659→                    ))}
   660→                  </div>
   661→                </CardContent>
   662→              </Card>
   663→            </>
   664→          ) : (
   665→            <Card>
   666→              <CardContent className="p-8 text-center text-muted-foreground">
   667→                {t('compute.no_jobs') || 'Select a job to view details'}
   668→              </CardContent>
   669→            </Card>
   670→          )}
   671→
   672→          {/* Node Status */}
   673→          <Card>
   674→            <CardHeader className="pb-3">
   675→              <CardTitle className="text-sm">{t('compute.nodes') || 'Compute Nodes'}</CardTitle>
   676→            </CardHeader>
   677→            <CardContent className="space-y-3">
   678→              {nodes.map(node => (
   679→                <div key={node.id} className="space-y-1">
   680→                  <div className="flex items-center justify-between text-sm">
   681→                    <div className="flex items-center gap-2">
   682→                      <span className={`w-2 h-2 rounded-full ${
   683→                        node.status === 'online' ? 'bg-green-500' :

   686→                      <span className="font-medium">{node.name}</span>

   690→                    </span>
