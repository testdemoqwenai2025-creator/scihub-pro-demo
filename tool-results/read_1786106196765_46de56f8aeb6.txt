   260→// ============ HELPER FUNCTIONS ============
   261→
   262→/**
   263→ * Creates a dynamic field with synthetic default value
   264→ */
   265→export function createDynamicField<T>(syntheticValue: T): DynamicField<T> {
   266→  return {
   267→    value: syntheticValue,
   268→    syntheticValue,
   269→    isDirty: false,
   270→    isValid: true,
   271→  };
   272→}
   273→
   274→/**
   275→ * Updates a dynamic field's value and marks as dirty if different from synthetic
   276→ */
   277→export function updateDynamicField<T>(field: DynamicField<T>, newValue: T): DynamicField<T> {
   278→  const isDirty = JSON.stringify(newValue) !== JSON.stringify(field.syntheticValue);
   279→  return {
   280→    ...field,
   281→    value: newValue,
   282→    isDirty,
   283→    lastModified: new Date(),
   284→    isValid: true,
   285→  };
   286→}
   287→
   288→/**
   289→ * Resets a dynamic field back to its synthetic value
   290→ */
   291→export function resetDynamicField<T>(field: DynamicField<T>): DynamicField<T> {
   292→  return {
   293→    ...field,
   294→    value: field.syntheticValue,
   295→    isDirty: false,
   296→    lastModified: new Date(),
   297→  };
   298→}

   300→/**
   301→ * Calculates approximate byte size of data
   302→ */
   303→function calculateDataSize(data: any): number {
   304→  return new Blob([JSON.stringify(data)]).size;
   305→}
