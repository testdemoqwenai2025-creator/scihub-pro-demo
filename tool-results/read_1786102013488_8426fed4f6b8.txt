   249→CTA #8: "Join Waitlist" (Email Form Submit)
   250→┌── Current Behavior ──────────────────────────────────────────────────────────┐

   252→│ • Current: Inside <form> tag but no onSubmit handler                          │

   254→└──────────────────────────────────────────────────────────────────────────────┘
   255→┌── REQUIRED BEHAVIOR ─────────────────────────────────────────────────────────┐
   256→│ ACTION: Capture Email & Add to Waitlist                                      │
   257→│                                                                              │
   258→│ IMPLEMENTATION:                                                              │
   259→│ □ Add onSubmit handler to form:                                              │
   260→│                                                                              │
   261→│   const handleWaitlistSubmit = async (e) => {                                │
   262→│     e.preventDefault();                                                      │
   263→│     const email = e.target.email.value;                                      │
   264→│                                                                              │
   265→│     // Validate email format                                                 │
   266→│     if (!email.includes('@')) {                                             │
   267→│       setError('Please enter a valid email');                                │
   268→│       return;                                                                │
   269→│     }                                                                       │
   270→│                                                                              │
   271→│     setLoading(true);                                                        │
   272→│                                                                              │
   273→│     try {                                                                   │
   274→│       // Call waitlist API                                                   │
   275→│       const res = await fetch('/api/waitlist', {                             │
   276→│         method: 'POST',                                                     │
   277→│         body: JSON.stringify({ email, source: 'landing_page' })             │
   278→│       });                                                                  │
   279→│                                                                              │
   280→│       if (res.ok) {                                                         │
   281→│         setSuccess(true); // Show success state                              │
   282→│         // Confetti animation optional                                       │
   283→│       } else {                                                              │
   284→│         setError('Something went wrong. Please try again.');                 │
   285→│       }                                                                     │
   286→│     } catch {                                                               │
   287→│       // Fallback: store locally, sync later                                 │
   288→│       localStorage.setItem('scihub_waitlist_email', email);                  │
   289→│       setSuccess(true); // Optimistic update                                 │
   290→│     } finally {                                                             │
   291→│       setLoading(false);                                                     │
   292→│     }                                                                       │
   293→│   };                                                                        │
   294→│                                                                              │
   295→│ □ Create /api/waitlist route if not exists:                                  │
   296→│   - Store emails in array/file                                              │
   297→│   - Deduplicate by email                                                    │
   298→│   - Return success/error                                                    │
   299→│   - Rate limit (max 5 submissions/hour per IP)                              │
   300→│                                                                              │
   301→│ □ UI States:                                                                │
   302→│   - Default: Email input + "Join Waitlist" button                            │
   303→│   - Loading: Spinner on button, disabled input                               │
   304→│   - Success: Green checkmark + "You're on the list! 🎉"                     │

  1658→export function ComingSoon({

  1683→  };
  1684→
  1685→  return (
  1686→    <Dialog>
  1687→      <DialogContent>
  1688→        <DialogTitle>🚧 {featureName}</DialogTitle>
  1689→        <div className="space-y-4">
  1690→          <p className="text-muted-foreground">
  1691→            This feature is under development.
  1692→          </p>
  1693→          
  1694→          <ul className="space-y-2">
  1695→            {benefits.map((b, i) => (

  1697→                <span className="text-green-500">✓</span> {b}
  1698→              </li>
  1699→            ))}
  1700→          </ul>
  1701→
  1702→          <div>
  1703→            <p className="text-sm font-medium mb-2">Progress</p>

  1710→          {!notified ? (
  1711→            <div className="space-y-2">
  1712→              <Input
  1713→                placeholder="your@email.com"
  1714→                value={email}
  1715→                onChange={(e) => setEmail(e.target.value)}
  1716→              />
  1717→              <Button 
  1718→                onClick={handleNotify} 
  1719→                disabled={!email || loading}
  1720→                className="w-full"
