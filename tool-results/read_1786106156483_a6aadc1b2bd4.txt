   664→function createVolcanoPlot(containerId) {
   665→  const container = document.getElementById(containerId);
   666→  
   667→  // Setup SVG
   668→  const width = 800;
   669→  const height = 500;
   670→  const margin = { top: 40, right: 40, bottom: 60, left: 70 };
   671→  
   672→  // Scales
   673→  const xScale = d3.scaleLinear()
   674→    .domain(d3.extent(data.genes, d => d.foldChange))
   675→    .range([margin.left, width - margin.right]);
   676→    
   677→  const yScale = d3.scaleLinear()
   678→    .domain([0, -Math.log10(d3.min(data.genes, d => d.pValue)) * 1.1])
   679→    .range([height - margin.bottom, margin.top]);
   680→  
   681→  // Draw points
   682→  const svg = d3.select(container).append('svg')
   683→    .attr('width', width)
   684→    .attr('height', height);
   685→    
   686→  svg.selectAll('circle')
   687→    .data(data.genes)
   688→    .enter()
   689→    .append('circle')
   690→    .attr('cx', d => xScale(d.foldChange))
   691→    .attr('cy', d => yScale(-Math.log10(d.pValue)))
   692→    .attr('r', d => d.significant ? 6 : 4)
   693→    .attr('fill', d => {
   694→      if (d.significant && d.foldChange > 0) return '#e41a1c'; // Red up
   695→      if (d.significant && d.foldChange < 0) return '#377eb8'; // Blue down
   696→      return '#999999'; // Gray not significant
   697→    })
   698→    .attr('opacity', 0.7)
   699→    .on('mouseover', showTooltip)
   700→    .on('mouseout', hideTooltip);
   701→    
   702→  // Threshold lines
   703→  svg.append('line')
   704→    .attr('x1', xScale(-1))
   705→    .attr('y1', margin.top)
   706→    .attr('x2', xScale(-1))
   707→    .attr('y2', height - margin.bottom)
   708→    .attr('stroke', 'green')
   709→    .attr('stroke-dasharray', '5,5');
   710→    
   711→  svg.append('line')
   712→    .attr('x1', xScale(1))
   713→    .attr('y1', margin.top)
   714→    .attr('x2', xScale(1))
   715→    .attr('y2', height - margin.bottom)
   716→    .attr('stroke', 'green')
   717→    .attr('stroke-dasharray', '5,5');
   718→    
   719→  svg.append('line')
   720→    .attr('x1', margin.left)
   721→    .attr('y1', yScale(-Math.log10(0.05)))
   722→    .attr('x2', width - margin.right)
   723→    .attr('y2', yScale(-Math.log10(0.05)))
   724→    .attr('stroke', 'blue')
   725→    .attr('stroke-dasharray', '5,5');
   726→

   729→    .attr('transform', \`translate(0, \${height - margin.bottom})\`)
   730→    .call(d3.axisBottom(xScale));

  1190→const generateMarkdownOutput = (code: string): string => {
  1191→  const outputs: string[] = [];
  1192→  
  1193→  outputs.push('');
  1194→  outputs.push('#'.repeat(60));
  1195→  outputs.push('# RENDERED MARKDOWN PREVIEW');
  1196→  outputs.push('#'.repeat(60));

  1199→  // Simple markdown rendering simulation
  1200→  const lines = code.split('\n');
  1201→  let inCodeBlock = false;
  1202→  
  1203→  for (const line of lines) {
  1204→    if (line.trim().startsWith('```')) {
  1205→      inCodeBlock = !inCodeBlock;
  1206→      if (inCodeBlock) {
  1207→        outputs.push('<div class="code-block">');
  1208→      } else {
  1209→        outputs.push('</div>');
  1210→      }
  1211→      continue;
  1212→    }
  1213→    
  1214→    if (inCodeBlock) {
  1215→      outputs.push('  ' + line);
  1216→      continue;
  1217→    }
  1218→    
  1219→    // Headers
  1220→    if (line.startsWith('# ')) {
  1221→      outputs.push(`<h1>${line.substring(2)}</h1>`);
  1222→    } else if (line.startsWith('## ')) {
  1223→      outputs.push(`<h2>${line.substring(3)}</h2>`);
  1224→    } else if (line.startsWith('### ')) {
  1225→      outputs.push(`<h3>${line.substring(4)}</h3>`);
  1226→    } else if (line.startsWith('> ')) {
  1227→      outputs.push(`<blockquote>${line.substring(2)}</blockquote>`);
  1228→    } else if (line.startsWith('- ') || line.startsWith('* ')) {
  1229→      outputs.push(`<li>${line.substring(2)}</li>`);
  1230→    } else if (line.match(/^\d+\.\s/)) {
  1231→      outputs.push(`<li>${line.replace(/^\d+\.\s/, '')}</li>`);
  1232→    } else if (line.trim() === '---') {
  1233→      outputs.push('<hr/>');
  1234→    } else if (line.trim()) {
  1235→      outputs.push(`<p>${line}</p>`);
  1236→    }
  1237→  }
  1238→  
  1239→  outputs.push('');
  1240→  outputs.push('#'.repeat(60));
  1241→  outputs.push('*Full Markdown rendering would include syntax highlighting,');
  1242→  outputs.push('table formatting, image embedding, and MathJax support*');
  1243→  outputs.push('#'.repeat(60));
