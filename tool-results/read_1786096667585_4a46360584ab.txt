   485→export function createDynamicField<T>(value: T, synthetic = true): DynamicField<T> {
   486→  return {
   487→    value,
   488→    syntheticValue: synthetic ? value : (undefined as T),
   489→    isDirty: false,
   490→    isValid: true,
   491→  };
   492→}
   493→
   494→export function updateDynamicField<T>(field: DynamicField<T>, newValue: T): DynamicField<T> {
   495→  return {
   496→    ...field,
   497→    value: newValue,
   498→    isDirty: JSON.stringify(newValue) !== JSON.stringify(field.syntheticValue),
   499→    lastModified: new Date(),
   500→    isValid: true,
   501→  };
   502→}
