   167→export default function ComputePage() {
   168→  const { t } = useTranslation();

   170→    createJob,
   171→    updateJobProgress,
   172→    cancelJob,
   173→    addActivity,
   174→  } = useDynamicStore();
   175→
   176→  // UI State
   177→  const [jobs, setJobs] = useState<ComputeJob[]>(INITIAL_JOBS);
   178→  const [nodes] = useState<ComputeNode[]>(INITIAL_NODES);

   181→  const [isLoading, setIsLoading] = useState(true);
   182→  
   183→  // Form state
   184→  const [newJobName, setNewJobName] = useState('');
   185→  const [newJobType, setNewJobType] = useState<ComputeJob['type']>('analysis');
   186→  const [newJobPriority, setNewJobPriority] = useState<ComputeJob['priority']>('normal');
   187→  const [newJobDataset, setNewJobDataset] = useState('');
   188→  const [newJobGpus, setNewJobGpus] = useState(1);
   189→
   190→  // Simulate loading on mount
   191→  useEffect(() => {
   192→    const timer = setTimeout(() => setIsLoading(false), 1500);
   193→    return () => clearTimeout(timer);
   194→  }, []);
   195→

   197→  useEffect(() => {
   198→    const interval = setInterval(() => {
   199→      setJobs(prev => prev.map(job => {

   201→          const newProgress = Math.min(100, job.progress + Math.random() * 2);
   202→          const newComputeHours = job.computeHoursUsed + (Math.random() * 0.1);
   203→          
   204→          return {
   205→            ...job,
   206→            progress: newProgress,
   207→            computeHoursUsed: newComputeHours,
   208→            ...(newProgress >= 100 ? { 
   209→              status: 'completed' as const, 
   210→              completedAt: new Date(),
   211→              estimatedCost: job.estimatedCost ? job.estimatedCost + Math.random() * 0.5 : undefined
   212→            } : {}),
   213→          };
   214→        }
   215→        return job;
   216→      }));
   217→    }, 3000);
   218→
   219→    return () => clearInterval(interval);
   220→  }, []);
   221→

   223→  const activeJobs = jobs.filter(j => j.status === 'running');
   224→  const totalCompute = jobs.reduce((acc, j) => acc + j.computeHoursUsed, 0);
   225→  const totalGPUs = activeJobs.reduce((acc, j) => acc + j.gpusAllocated, 0);
   226→  const avgProgress = activeJobs.length > 0 
   227→    ? Math.round(activeJobs.reduce((acc, j) => acc + j.progress, 0) / activeJobs.length)
   228→    : 0;
   229→

   231→  const handleCreateJob = () => {
   232→    if (!newJobName.trim()) return;
   233→
   234→    const costPerHour = newJobGpus * 0.5; // $0.50 per GPU hour (free tier simulation)

   236→
   237→    const newJob: ComputeJob = {
   238→      id: `job-${Date.now()}`,
   239→      name: newJobName,
   240→      type: newJobType,
   241→      status: 'queued',
   242→      priority: newJobPriority,
   243→      submitter: 'You',
   244→      progress: 0,
   245→      computeHoursUsed: 0,
   246→      computeHoursTotal: estimatedHours,
   247→      gpusAllocated: 0,
   248→      memoryUsed: '0 GB',
   249→      memoryTotal: `${newJobGpus * 16} GB`,
   250→      submittedAt: new Date(),
   251→      estimatedCost: estimatedHours * costPerHour,
   252→      logs: [{ timestamp: new Date(), level: 'INFO', message: 'Job created and queued' }],
   253→    };
   254→
   255→    setJobs(prev => [newJob, ...prev]);
   256→    
   257→    // Also add to store
   258→    createJob({
   259→      name: createDynamicField(newJobName),
   260→      type: createDynamicField(newJobType),
   261→      status: createDynamicField('queued'),
   262→      priority: createDynamicField(newJobPriority),
   263→      submitter: createDynamicField('You'),
   264→      config: {
   265→        inputDataset: createDynamicField(newJobDataset),
   266→        parameters: {},
   267→        outputFormat: createDynamicField('JSON'),
   268→        notifications: createDynamicField(true),
   269→        retryCount: createDynamicField(3),
   270→      },
   271→      progress: createDynamicField(0),
   272→      computeHoursUsed: createDynamicField(0),
   273→      computeHoursTotal: createDynamicField(estimatedHours),
   274→      gpusAllocated: createDynamicField(0),
   275→      memoryUsed: createDynamicField('0 GB'),
   276→    });

   278→    addActivity({
   279→      type: 'job',
   280→      message: createDynamicField(`Submitted job "${newJobName}"`),
   281→      icon: '⚡',
   282→    });
   283→
   284→    // Reset form
   285→    setNewJobName('');
   286→    setShowCreateForm(false);
   287→  };

   292→    if (!job) return;
   293→
   294→    if (confirm(`Cancel job "${job.name}"?`)) {
   295→      setJobs(prev => prev.map(j => 
   296→        j.id === jobId ? { ...j, status: 'cancelled' as const } : j
   297→      ));
   298→      cancelJob(jobId);
   299→      
   300→      addActivity({
   301→        type: 'update',
   302→        message: createDynamicField(`Cancelled job "${job.name}"`),
   303→        icon: '🛑',
   304→      });
   305→    }
   306→  };
   307→

   312→      case 'queued': return <Badge className="bg-yellow-500">{t('compute.queued') || 'Queued'}</Badge>;
   313→      case 'completed': return <Badge className="bg-green-500">{t('compute.completed') || 'Completed'}</Badge>;

   317→    }
   318→  };

   324→      normal: 'bg-blue-400',
   325→      high: 'bg-orange-400',
   326→      urgent: 'bg-red-500',
   327→    };

   329→  };

   332→  const formatTimeAgo = (date: Date): string => {
   333→    const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
   334→    if (seconds < 60) return `${seconds}s ago`;
   335→    const minutes = Math.floor(seconds / 60);
   336→    if (minutes < 60) return `${minutes}m ago`;
   337→    const hours = Math.floor(minutes / 60);
   338→    if (hours < 24) return `${hours}h ago`;
   339→    return `${Math.floor(hours / 24)}d ago`;
   340→  };
   341→
   342→  // Loading skeleton
   343→  if (isLoading) {
   344→    return (
   345→      <div className="min-h-screen bg-background">
   346→        <ComputeSkeleton />
   347→      </div>
   348→    );
   349→  }
   350→
   351→  return (
   352→    <div className="min-h-screen bg-background p-6">
   353→      {/* Header */}
   354→      <div className="mb-6">
   355→        <h1 className="text-3xl font-bold text-foreground">{t('compute.title') || 'Compute Layer'}</h1>
   356→        <p className="text-muted-foreground mt-1">
   357→          Manage computational jobs and monitor cluster resources. Free-tier simulation included.
   358→        </p>
   359→        
   360→        <div className="mt-2 flex items-center gap-3">
   361→          <Badge variant="secondary" className="bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-200">
   362→            🆓 Free Tier Compute
   363→          </Badge>
   364→          <span className="text-sm text-muted-foreground">
   365→            Simulated resources • No actual costs incurred
   366→          </span>
   367→        </div>
   368→      </div>
   369→
   370→      {/* Stats Grid */}
   371→      <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-6">
   372→        <Card>
   373→          <CardContent className="p-4 flex items-center gap-3">
   374→            <span className="text-2xl">⚡</span>
   375→            <div>
   376→              <p className="text-sm text-muted-foreground">{t('compute.active_jobs') || 'Active Jobs'}</p>
   377→              <p className="text-xl font-bold text-blue-500">{activeJobs.length}</p>
   378→            </div>
   379→          </CardContent>
   380→        </Card>
   381→        <Card>
   382→          <CardContent className="p-4 flex items-center gap-3">
   383→            <span className="text-2xl">⏱️</span>
   384→            <div>
   385→              <p className="text-sm text-muted-foreground">Total Compute Used</p>
   386→              <p className="text-xl font-bold">{totalCompute.toFixed(1)} hrs</p>
   387→            </div>
   388→          </CardContent>
   389→        </Card>
   390→        <Card>
   391→          <CardContent className="p-4 flex items-center gap-3">
   392→            <span className="text-2xl">🖥️</span>
   393→            <div>
   394→              <p className="text-sm text-muted-foreground">GPUs Allocated</p>
   395→              <p className="text-xl font-bold text-purple-500">{totalGPUs}</p>
   396→            </div>
   397→          </CardContent>
   398→        </Card>
   399→        <Card>
   400→          <CardContent className="p-4 flex items-center gap-3">
   401→            <span className="text-2xl">📊</span>
   402→            <div>
   403→              <p className="text-sm text-muted-foreground">Avg Progress</p>
   404→              <p className="text-xl font-bold text-green-500">{avgProgress}%</p>
   405→            </div>
   406→          </CardContent>
   407→        </Card>
   408→      </div>
   409→
   410→      {/* Create Job Button */}
   411→      <div className="mb-6 flex justify-between items-center">
   412→        <h2 className="text-xl font-semibold">{t('compute.jobs') || 'Jobs'}</h2>
   413→        <Button onClick={() => setShowCreateForm(!showCreateForm)}>
   414→          + {t('compute.new_job') || 'New Job'}
   415→        </Button>
   416→      </div>
   417→
   418→      {/* Create Job Form */}
   419→      {showCreateForm && (

   421→          <CardHeader>
   422→            <CardTitle className="text-base">Submit New Job</CardTitle>

   432→                  className="mt-1"
   433→                />
   434→              </div>
   435→              <div>

   476→                  </SelectContent>
   477→                </Select>
   478→              </div>
   479→              <div className="md:col-span-2">

   482→                  value={newJobDataset}
   483→                  onChange={(e) => setNewJobDataset(e.target.value)}

   485→                  className="mt-1"

   487→              </div>

   490→            {/* Cost Estimate */}

   512→
   513→      <div className="flex gap-6">

   516→          {jobs.map(job => (

   522→              onClick={() => setSelectedJob(job)}
   523→            >
